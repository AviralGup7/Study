\spsection{Part A -- Mole balances and rates (Ch1-krd (1).pdf)}

\qtitle{Q1 -- Rocket engine: rates of reaction}{\label{q:1}Ch1, slide 36}
\src{Ch1-krd (1).pdf, slide 36 (the ``Example'' slide).}
A rocket engine burns a stoichiometric mixture of fuel (liquid hydrogen) in oxidant
(liquid oxygen).  The combustion chamber is cylindrical, 75\,cm long and 60\,cm in
diameter, and the combustion process produces 108\,kg/s of exhaust gases.  If
combustion is complete, find the rate of reaction of hydrogen and of oxygen.

\begin{qbox}
\textbf{What ``rate of reaction'' means here.}  The rate is defined per unit
extensive quantity of the system (Ch1 slides 30--34); for a homogeneous gas-phase
combustion chamber the natural basis is the reactor \emph{volume},
$-r_i = (\text{moles of } i \text{ reacting per unit time})/V$, with units
mol/(m$^3\cdot$s).  So the whole problem is: find the molar rates at which
H$_2$ and O$_2$ are consumed, and divide by the chamber volume.

\textbf{Step 1 --- the chamber volume.}  A cylinder of length $L=0.75$~m and
diameter $D=0.60$~m:
\[
V = \frac{\pi}{4}D^2 L = \frac{\pi}{4}(0.60)^2(0.75) = 0.2121\ \mathrm{m^3}.
\]

\textbf{Step 2 --- turn the exhaust mass flow into a molar flow.}  Complete
combustion means every gram of exhaust leaves as water,
$\mathrm{H_2 + \tfrac12 O_2 \to H_2O}$, of molar mass $18$~kg/kmol:
\[
F_{\mathrm{H_2O}} = \frac{108\ \mathrm{kg/s}}{18\ \mathrm{kg/kmol}} = 6.0\ \mathrm{kmol/s}.
\]

\textbf{Step 3 --- read the reactant rates off the stoichiometry.}  One mole of
H$_2$ and half a mole of O$_2$ disappear per mole of water formed, so
\[
F_{\mathrm{H_2,\,reacted}} = 6.0\ \mathrm{kmol/s},\qquad
F_{\mathrm{O_2,\,reacted}} = \tfrac12(6.0) = 3.0\ \mathrm{kmol/s}.
\]

\textbf{Step 4 --- divide by the volume.}  At steady state with complete, uniform
combustion the mole balance In $-$ Out $+$ Generation $=$ Accumulation$(=0)$ gives
the consumption rate per unit volume directly:
\[
-r_{\mathrm{H_2}} = \frac{6.0}{0.2121} = 28.3\ \mathrm{kmol\,m^{-3}\,s^{-1}},\qquad
-r_{\mathrm{O_2}} = \frac{3.0}{0.2121} = 14.1\ \mathrm{kmol\,m^{-3}\,s^{-1}}.
\]
\end{qbox}
\note{Sanity checks for the exam: the two rates are in the same $2{:}1$ ratio as the
stoichiometric coefficients (this is the relative-rate rule of Part~C), and the
magnitudes sit in the ``rocket engines'' band of the rate spectrum printed on
slide~35 --- combustion is among the fastest reactions there are.}
\begin{ansbox}$-r_{\mathrm{H_2}} \approx 28.3$ kmol/(m$^3\cdot$s) $\approx 2.8\times10^{4}$ mol/(m$^3\cdot$s);
$-r_{\mathrm{O_2}} \approx 14.1$ kmol/(m$^3\cdot$s) $\approx 1.4\times10^{4}$ mol/(m$^3\cdot$s).\end{ansbox}

\qtitle{Q2 -- CSTR, PFR and batch for three rate laws}{\label{q:2}Ch1, slide 54}
\src{Ch1-krd (1).pdf, slide 54 (the ``Problem'' slide).}
The reaction $A\to B$ is to be carried out isothermally in a continuous-flow
reactor.  Calculate both the CSTR and the PFR volumes necessary to consume 99\%
of A ($C_A = 0.01\,C_{A0}$) when the entering molar flow rate is $F_{A0}=5$~mol/h,
assuming the rate laws (a) $-r_A=k$ with $k=0.05$~mol/(h$\cdot$dm$^3$),
(b) $-r_A=kC_A$ with $k=0.0001$~s$^{-1}$, (c) $-r_A=kC_A^2$ with
$k=3$~dm$^3$/(mol$\cdot$h).  The entering volumetric flow rate is
$v_0=10$~dm$^3$/h.  (d)~Repeat each case for a batch reactor reaching 99\%.

\begin{qbox}
\textbf{Set-up used for every part.}  $X = 0.99$ and
$C_{A0} = F_{A0}/v_0 = 5/10 = 0.5$~mol/dm$^3$.  The three design equations
(slides 42--53):
\[
\text{CSTR: } V = \frac{F_{A0}\,X}{(-r_A)_{\text{exit}}},\qquad
\text{PFR: } V = F_{A0}\int_0^X \frac{dX}{-r_A},\qquad
\text{batch: } t = N_{A0}\int_0^X \frac{dX}{-r_A\,V} = C_{A0}\int_0^X \frac{dX}{-r_A}\ (V\ \text{const}).
\]
For a flow reactor with $-r_A = kC_A^n$ and $C_A = C_{A0}(1-X)$, the integrals
below are the standard ones; write them out in the exam rather than quoting results.

\textbf{(a) Zero order, $-r_A = k = 0.05$.}  The rate is constant, so the CSTR
exit rate equals the rate everywhere in the PFR and the two reactors coincide:
\[
V_{CSTR} = \frac{F_{A0}X}{k} = \frac{5(0.99)}{0.05} = 99\ \mathrm{dm^3},\qquad
V_{PFR} = \frac{F_{A0}}{k}\int_0^{0.99} dX = 99\ \mathrm{dm^3}.
\]
Batch: $t = \dfrac{C_{A0}}{k}\displaystyle\int_0^{0.99} dX = \dfrac{0.5(0.99)}{0.05} = 9.9$~h.

\textbf{(b) First order, $-r_A = kC_A$.}  First fix the units:
$k = 0.0001\ \mathrm{s^{-1}} \times 3600 = 0.36\ \mathrm{h^{-1}}$ --- the single
most common slip in this question.  With $C_A = C_{A0}(1-X)$:
\begin{align*}
V_{CSTR} &= \frac{F_{A0}X}{kC_{A0}(1-X)} = \frac{v_0 X}{k(1-X)}
 = \frac{10(0.99)}{0.36(0.01)} = 2750\ \mathrm{dm^3},\\
V_{PFR} &= \frac{F_{A0}}{kC_{A0}}\int_0^X\frac{dX}{1-X}
 = \frac{v_0}{k}\ln\frac{1}{1-X} = \frac{10}{0.36}(4.605) = 127.9\ \mathrm{dm^3},\\
t_{batch} &= \frac{1}{k}\ln\frac{1}{1-X} = \frac{4.605}{0.36} = 12.8\ \mathrm{h}.
\end{align*}

\textbf{(c) Second order, $-r_A = kC_A^2 = kC_{A0}^2(1-X)^2$.}
\begin{align*}
V_{CSTR} &= \frac{F_{A0}X}{kC_{A0}^2(1-X)^2} = \frac{v_0 X}{kC_{A0}(1-X)^2}
 = \frac{10(0.99)}{3(0.5)(0.01)^2} = 66\,000\ \mathrm{dm^3},\\
V_{PFR} &= \frac{v_0}{kC_{A0}}\int_0^X\frac{dX}{(1-X)^2}
 = \frac{v_0}{kC_{A0}}\cdot\frac{X}{1-X} = \frac{10}{1.5}(99) = 660\ \mathrm{dm^3},\\
t_{batch} &= \frac{1}{kC_{A0}}\cdot\frac{X}{1-X} = \frac{99}{1.5} = 66\ \mathrm{h}.
\end{align*}
\end{qbox}
\note{What the numbers teach (and what the deck's next section turns into the
Levenspiel plot): as the order rises the CSTR volume explodes ---
99 $\to$ 2750 $\to$ 66\,000~dm$^3$ --- while the PFR grows mildly ---
99 $\to$ 128 $\to$ 660~dm$^3$.  The CSTR operates at the \emph{exit} rate, i.e.\ at
the smallest rate in the reactor, and punishing a small rate with a big volume is
exactly what the design equations do.  At 99\% conversion a CSTR is 100$\times$ the
PFR for the second-order case.}
\begin{ansbox}(a) 99 dm$^3$ (CSTR $=$ PFR), batch 9.9 h; (b) CSTR 2750, PFR 127.9 dm$^3$, batch 12.8 h;
(c) CSTR 66\,000, PFR 660 dm$^3$, batch 66 h.\end{ansbox}

\spsection{Part B -- Conversion and reactor sizing (Ch2-krd (1).pdf)}

\qtitle{Q3 -- Sizing from the Levenspiel table}{\label{q:3}Ch2, slides 19--24}
\src{Ch2-krd (1).pdf, slides 19--24 (the worked ``CSTR/PFR sizing'' problem).}
Consider the isothermal gas-phase isomerization $A\to B$ ($T=500$~K,
$P=830$~kPa $=8.2$~atm, charge pure A).  The measured rates are:
\begin{center}\small
\begin{tabular}{lccccccc}
\toprule
$X$ & 0 & 0.1 & 0.2 & 0.4 & 0.6 & 0.7 & 0.8\\
$-r_A$ (mol/m$^3\cdot$s) & 0.45 & 0.37 & 0.30 & 0.195 & 0.113 & 0.079 & 0.05\\
\bottomrule
\end{tabular}
\end{center}
Calculate the volumes required to achieve $80$\% conversion in a CSTR and in a
PFR when species A enters at $F_{A0}=0.4$~mol/s.

\begin{qbox}
\textbf{Step 0 --- build the Levenspiel column.}  The deck's method (slides
10--12) sizes reactors from the plot of $F_{A0}/(-r_A)$ vs $X$: the CSTR volume is
a \emph{rectangle} (height at exit conversion times width $X$); the PFR volume is
the \emph{area under the curve}.  Dividing $F_{A0}=0.4$~mol/s by each rate gives
\begin{center}\small
\begin{tabular}{lccccccc}
\toprule
$X$ & 0 & 0.1 & 0.2 & 0.4 & 0.6 & 0.7 & 0.8\\
$F_{A0}/(-r_A)$ (m$^3$) & 0.89 & 1.08 & 1.33 & 2.05 & 3.54 & 5.06 & 8.00\\
\bottomrule
\end{tabular}
\end{center}

\textbf{Step 1 --- CSTR volume (slide 20).}  A CSTR operates at its \emph{exit}
conditions, so evaluate the rate at $X=0.8$: $\left(1/-r_A\right)_{0.8}=1/0.05=20$
m$^3\cdot$s/mol, i.e.\ $F_{A0}/(-r_A)=8.00$~m$^3$, and
\[
V_{\mathrm{CSTR}}=\left(\frac{F_{A0}}{-r_A}\right)_{X=0.8}X
=0.4\ \tfrac{\mathrm{mol}}{\mathrm s}\times20\ \tfrac{\mathrm{m^3\,s}}{\mathrm{mol}}\times0.8
=6.4\ \mathrm{m^3}.
\]

\textbf{Step 2 --- PFR volume by Simpson's rule (slides 21--23).}
$V_{\mathrm{PFR}}=\int_0^{0.8}\frac{F_{A0}}{-r_A}\,dX$ has no closed form without a
rate law, so the deck applies its five-point quadrature with
$h=\Delta X=(0.8-0)/4=0.2$:
\[
V=\frac{\Delta X}{3}\Big[\tfrac{F_{A0}}{-r_A}(0)+4\tfrac{F_{A0}}{-r_A}(0.2)
+2\tfrac{F_{A0}}{-r_A}(0.4)+4\tfrac{F_{A0}}{-r_A}(0.6)+\tfrac{F_{A0}}{-r_A}(0.8)\Big]
\]
\[
=\frac{0.2}{3}\big[0.89+4(1.33)+2(2.05)+4(3.54)+8.00\big]
=\frac{0.2}{3}(32.47)=2.165\ \mathrm{m^3}.
\]

\textbf{Step 3 --- read the picture (slide 24).}  The CSTR rectangle minus the
PFR area is $6.4-2.165=4.235$~m$^3$ of extra volume --- the price of running the
whole reactor at the slow exit rate.
\end{qbox}
\note{Exam habit: write the two design equations, the $F_{A0}/(-r_A)$ table, the
rectangle-vs-area argument, and the Simpson arithmetic.  The deck's own printed
answers are exactly $6.4$ and $2.165$~m$^3$.}
\begin{ansbox}$V_{CSTR}=6.4$ m$^3$, $V_{PFR}=2.165$ m$^3$ (the PFR is $4.235$ m$^3$ smaller).\end{ansbox}

\qtitle{Q4 -- Two CSTRs in series}{\label{q:4}Ch2, slides 31--36}
\src{Ch2-krd (1).pdf, slides 35--36.}
For two CSTRs in series, $40$\% conversion is achieved in the first reactor.
What is the volume of each reactor necessary to achieve $80$\% overall conversion
of the entering species?  (Same feed and rate data as Q3.)

\begin{qbox}
\textbf{The design equation for the $n$-th CSTR (slides 32--34).}  Balance A over
reactor 1: In $-$ Out $+$ Generation $=0$, i.e.\ $F_{A0}-F_{A1}+r_{A1}V_1=0$; with
$F_{A1}=F_{A0}-F_{A0}X_1$ this gives $V_1=F_{A0}\left(\frac{1}{-r_{A1}}\right)X_1$.
Over reactor 2 the balance is $F_{A1}-F_{A2}+r_{A2}V_2=0$; substituting
$F_{A1}=F_{A0}(1-X_1)$ and $F_{A2}=F_{A0}(1-X_2)$ leaves the difference of
conversions:
\[
V_2=\frac{F_{A1}-F_{A2}}{-r_{A2}}=\frac{F_{A0}}{-r_{A2}}\,(X_2-X_1).
\]
Graphically each CSTR is a rectangle whose height is its \emph{own exit} value of
$F_{A0}/(-r_A)$.

\textbf{Step 1 --- reactor 1 ($X_1=0.4$, slide 36).}
$F_{A0}/(-r_A)\big|_{0.4}=2.05$~m$^3$:
\[
V_1 = 2.05\ \mathrm{m^3}\times0.4 = 0.82\ \mathrm{m^3}.
\]

\textbf{Step 2 --- reactor 2 ($X_1=0.4 \to X_2=0.8$).}
$F_{A0}/(-r_A)\big|_{0.8}=8.00$~m$^3$:
\[
V_2 = 8.00\ \mathrm{m^3}\times(0.8-0.4) = 3.20\ \mathrm{m^3}.
\]

\textbf{Step 3 --- total and comparison.}
$V=V_1+V_2=0.82+3.20=4.02$~m$^3$, against $6.4$~m$^3$ for a single CSTR to 80\%:
splitting the conversion across two tanks saves $2.38$~m$^3$, because the first
tank never has to operate at the slow $X=0.8$ rate.
\end{qbox}
\note{Exam habit: sketch the two Levenspiel rectangles (short one on $0\le X\le0.4$,

\qtitle{Q5 -- Two PFRs in series}{\label{q:5}Ch2, slides 37--39}
\src{Ch2-krd (1).pdf, slides 37--39.}
Calculate the reactor volumes $V_1$ and $V_2$ for the plug-flow sequence shown
when the intermediate conversion is $40$\% and the final conversion is $80$\%
(same feed as Q3).

\begin{qbox}
\textbf{Why each PFR integrates over its own slice.}  Reactor 1 raises $X$ from
$0$ to $0.4$ and reactor 2 from $0.4$ to $0.8$; each integrates $F_{A0}/(-r_A)$
over that interval, so the two areas simply tile the area of one PFR (slide 44
proves that $N$ PFRs in series give the same conversion as a single PFR of the
total volume).

\textbf{Step 1 --- reactor 1 (slide 38), three-point Simpson, $h=0.2$:}
\[
V_1=\int_0^{0.4}\frac{F_{A0}}{-r_A}\,dX
=\frac{0.2}{3}\big[0.89+4(1.33)+2.05\big]
=\frac{0.2}{3}(8.26)=0.551\ \mathrm{m^3}.
\]

\textbf{Step 2 --- reactor 2 (slide 39), points $0.4,\,0.6,\,0.8$:}
\[
V_2=\int_{0.4}^{0.8}\frac{F_{A0}}{-r_A}\,dX
=\frac{0.2}{3}\big[2.05+4(3.54)+8.00\big]

\qtitle{Q6 -- Mixed reactor, unknown stoichiometry}{\label{q:6}Ch2, slides 41--42}
\src{Ch2-krd (1).pdf, slides 42--43 (the worked mixed-reactor problem).}
One liter per minute of liquid containing A and B ($C_{A0}=0.10$, $C_{B0}=0.01$
mol/liter) flows into a mixed reactor of volume $V=1$~liter.  The materials react
in a complex manner for which the stoichiometry is unknown.  The outlet stream
contains A, B and C with $C_A=0.02$, $C_B=0.03$, $C_C=0.04$ mol/liter.  Find the
rate of reaction of A, B and C for the conditions within the reactor.

\begin{qbox}
\textbf{Why one balance per species.}  A CSTR at steady state balances each
species on its own: In $-$ Out $+$ Generation $=0$,
$v_0C_{i0}-v_0C_i+r_iV=0$, so with the space time $\tau=V/v_0=1/1=1$~min
\[
r_i=\frac{C_i-C_{i0}}{\tau}.
\]
Because the stream is liquid (constant volume), every species can be balanced
independently --- which is exactly why the question can be solved even though
the stoichiometry is unknown.

\textbf{Step 1 --- evaluate the three balances.}
\[
r_A=\frac{0.02-0.10}{1}=-0.08,\qquad
r_B=\frac{0.03-0.01}{1}=+0.02,\qquad
r_C=\frac{0.04-0}{1}=+0.04\quad\mathrm{mol/(L\cdot min)}.
\]
A is consumed; B and C are produced.

\textbf{Step 2 --- the ratios reveal the stoichiometry.}  Rates of reaction are
proportional to the stoichiometric coefficients (the relative-rate rule of
Part~C): $-r_A:r_B:r_C=0.08:0.02:0.04=4:1:2$, so the equation consistent with
the measurements is $4A\to B+2C$.
\end{qbox}
\note{Exam phrasing: ``the reactor is a CSTR, hence each species obeys
$r_i=(C_i-C_{i0})/\tau$; the signs give consumption/production and the ratio of
the rates gives the stoichiometric coefficients.''}
\begin{ansbox}$-r_A=0.08$, $r_B=0.02$, $r_C=0.04$ mol/(L$\cdot$min); the data imply $4A\to B+2C$.\end{ansbox}

\qtitle{Q7 -- PFRs in two parallel branches}{\label{q:7}Ch2, slides 45--46}
\src{Ch2-krd (1).pdf, slides 45--46.}
The reactor setup consists of three plug-flow reactors in two parallel branches:
branch D has a reactor of volume 50 liters followed by a reactor of volume
30 liters; branch E has a reactor of volume 40 liters.  What fraction of the
feed should go to branch D?

\begin{qbox}
\textbf{The rule (slide 45).}  PFRs in any parallel / parallel-series combination
behave as a single PFR of the total volume \emph{only if} the feed is distributed
so that streams which meet have the same composition --- i.e.\ the space time
$V/F$ must be the same for each parallel line; ``any other way of feeding is less
efficient''.

\textbf{Step 1 --- branch volumes.}  Series PFRs add (slide 44): branch~D is
$50+30=80$~L; branch~E is $40$~L.

\textbf{Step 2 --- equal space times.}  $V_D/F_D=V_E/F_E \Rightarrow
F_D/F_E=80/40=2$.

\textbf{Step 3 --- the fraction.}
\[
\frac{F_D}{F}=\frac{80}{80+40}=\frac{2}{3}\approx0.67.
\]
\end{qbox}
\note{An earlier revision of these notes misread the figure as $20+30$~L and
printed $5/9$; the slide prints 50 then 30 liters, so two-thirds is the value
consistent with the deck.}
\begin{ansbox}$2/3\approx0.67$ of the feed to branch D (equal space time $80/80=40/40=1$~L per unit flow).\end{ansbox}

\qtitle{Q8 -- P2-7: adiabatic $A+B\to C$, parts (a)--(e)}{\label{q:8}Ch2, slide 59}
\src{Ch2-krd (1).pdf, slides 59--60 (``P2-7'').}
The exothermic reaction $A\to B+C$ was carried out adiabatically and the
following data were recorded; the entering molar flow rate of A was
$F_{A0}=300$~mol/min:
\begin{center}\small
\begin{tabular}{lcccccccc}
\toprule
$X$ & 0 & 0.2 & 0.4 & 0.45 & 0.5 & 0.6 & 0.8 & 0.9\\
$-r_A$ (mol/dm$^3\cdot$min) & 1.0 & 1.67 & 5.0 & 5.0 & 5.0 & 5.0 & 1.25 & 0.91\\
\bottomrule
\end{tabular}

\qtitle{Q9 -- Relative rates: NO oxidation}{\label{q:9}Ch3, slides 5--7}
\src{Ch3-krd.pdf, slides 5--7 (``Example'').}
For $2\mathrm{NO}+\mathrm{O_2}\to2\mathrm{NO_2}$, NO$_2$ is formed at
$4$~mol/(m$^3\cdot$s).  What are the rates of reaction of NO and of O$_2$?

\begin{qbox}
\textbf{The rule (slide 5).}  For $aA+bB\to cC+dD$ the stoichiometry ties every
species rate together:
\[
\frac{r_A}{-a}=\frac{r_B}{-b}=\frac{r_C}{c}=\frac{r_D}{d},
\]
i.e.\ divide each rate by its own coefficient (taken negative for reactants) and
all four ratios equal the same single ``rate of reaction''.  Sign convention:
$r_i>0$ means formation, $r_i<0$ consumption.

\textbf{Step 1 --- write the chain for this reaction} ($a_{NO}=2$, $b_{O_2}=1$,
$c_{NO_2}=2$):
\[
\frac{r_{\mathrm{NO}}}{-2}=\frac{r_{\mathrm{O_2}}}{-1}=\frac{r_{\mathrm{NO_2}}}{+2}.
\]

\textbf{Step 2 --- scale by the known rate.}  With $r_{\mathrm{NO_2}}=+4$:
\[
r_{\mathrm{NO}}=\frac{-2}{+2}(4)=-4\ \mathrm{mol/(m^3\cdot s)},\qquad
r_{\mathrm{O_2}}=\frac{-1}{+2}(4)=-2\ \mathrm{mol/(m^3\cdot s)}.
\]

\textbf{Step 3 --- check.}  $|r_{NO}|:|r_{O_2}|:|r_{NO_2}|=4:2:4=2:1:2$ --- the
coefficients themselves, as the rule demands (this is the deck's solution slide 7).
\end{qbox}
\note{In the exam also state $-r_{\mathrm{NO}}=4$ mol/(m$^3\cdot$s): ``NO is
consumed as fast as NO$_2$ is formed; O$_2$ at half that rate.''}
\begin{ansbox}$r_{\mathrm{NO}}=-4$, $r_{\mathrm{O_2}}=-2$ mol/(m$^3\cdot$s).\end{ansbox}

\qtitle{Q10 -- Relative rates: $2A+3B\to5C$}{\label{q:10}Ch3, slides 6--7}
\src{Ch3-krd.pdf, slides 6--7 (``Exercise'').}

\begin{ansbox}$C_A=C_{A0}(1-X)$, $C_B=C_{A0}(\Theta_B-X/3)$, $C_C=C_{A0}(\Theta_C+X)$, $C_D=C_{A0}(\Theta_D+X/3)$.\end{ansbox}

\qtitle{Q14 -- Phosphine cracking PFR}{\label{q:14}Ch3, slide 63}
\src{Ch3-krd.pdf, slide 63 (``Example'').}
$4\,\mathrm{PH_3}(g)\to\mathrm{P_4}(g)+6\,\mathrm{H_2}(g)$ proceeds at
$649\,^\circ$C with the first-order rate
$-r_{\mathrm{PH_3}}=\left(\frac{10}{\mathrm{hr}}\right)C_{\mathrm{PH_3}}$.  What
size of plug-flow reactor operating at $649\,^\circ$C and $460$~kPa can produce
$80$\% conversion of a feed of $40$~mol/h of pure phosphine?

\begin{qbox}
\textbf{Step 1 --- enter in concentrations.}  Ideal gas at the reactor state:
\[
C_{A0}=\frac{P_0}{RT}=\frac{460\,000}{8.314\times922.15}=60.0\ \mathrm{mol/m^3}.
\]

\textbf{Step 2 --- the expansion factor.}  $4\to10$ moles:
$\varepsilon_A=\frac{10-4}{4}=1.5$, so
$C_A=C_{A0}\frac{1-X}{1+1.5X}$ --- the gas \emph{expands} as it cracks and the
concentration falls faster than $(1-X)$.

\textbf{Step 3 --- PFR design equation, first order.}
\[
V=F_{A0}\int_0^{0.8}\frac{dX}{kC_{A0}}\cdot\frac{1+1.5X}{1-X}
=\frac{F_{A0}}{kC_{A0}}\Big[(1+\varepsilon)\ln\frac{1}{1-X}-\varepsilon X\Big].
\]

\textbf{Step 4 --- substitute.}  $\frac{F_{A0}}{kC_{A0}}=\frac{40}{10\times60}=0.0667$~m$^3$;
the bracket at $X=0.8$: $2.5\ln5-1.5(0.8)=4.024-1.2=2.824$:
\[
V=0.0667\times2.824=0.188\ \mathrm{m^3}\ (\approx188\ \mathrm{L}).
\]
\end{qbox}
\note{Two classic slips the deck warns about implicitly: forgetting $\varepsilon$
(which would give $0.0667\times\ln5=0.107$~m$^3$, far too small) and using
$C_{A0}$ at STP instead of at $649\,^\circ$C/$460$~kPa.}
\begin{ansbox}$V\approx0.19$ m$^3$ (about 190 L).\end{ansbox}
