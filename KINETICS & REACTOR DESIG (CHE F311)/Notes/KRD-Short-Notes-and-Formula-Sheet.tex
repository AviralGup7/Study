\spsection{1 \enspace Mole Balances, Rates of Reaction and Conversion}
\notetext{Source: \texttt{Ch1-krd (1).pdf}.}

\spsub{1.1 \enspace Rate of reaction and the bases it can be written on}
\begin{itemize}
\item \textbf{Rate of reaction} is the number of moles of a species reacting per unit time per unit \emph{extensive quantity} of the system. The extensive quantity is what distinguishes the bases:
\end{itemize}
\begin{formulabox}
\begin{align*}
-r_A &= \text{mol}/(\text{volume}\cdot\text{time}) &&\text{homogeneous: per reactor volume } V\\
-r_A' &= \text{mol}/(\text{mass cat.}\cdot\text{time}) &&\text{heterogeneous: per catalyst mass } W\\
r_A'' &= \text{mol}/(\text{area}\cdot\text{time}) &&\text{per catalyst surface area } S\\[4pt]
-r_A' W &= r_A'' S = -r_A V
 &&\Longrightarrow\quad \boxed{\;\text{all three give the same molar rate}\;}
\end{align*}
\end{formulabox}
\notetext{Choosing the wrong basis is the most common error in packed-bed work: the design equation must be written on the same basis as the rate law.}

\spsub{1.2 \enspace Conversion, extent and the general mole balance}
\begin{formulabox}
\begin{align*}
X &= \frac{\text{moles of A reacted}}{\text{moles of A fed}}
 &&\text{conversion, defined on the limiting reactant}\\[3pt]
N_i &= N_i^{0} + \nu_i\,\alpha
 &&\text{extent of reaction } \alpha\\[3pt]
F_{j0} - F_j + G_j &= \frac{dN_j}{dt}
 &&\text{general mole balance on species } j
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{$G_j$} is the rate of generation of $j$ by reaction within the system volume --- $G_j = r_j V$. Everything else in the balance is a transport term.
\item Conversion is always defined on the \textbf{limiting} reactant, so that $0 \le X \le 1$.
\end{itemize}

\spsub{1.3 \enspace The four design equations}
\begin{formulabox}
\begin{align*}
\text{Batch:}\quad & \frac{dN_A}{dt} = r_A V
 &&\Longrightarrow\quad t = N_{A0}\int_0^X \frac{dX}{-r_A V}\\[6pt]
\text{CSTR:}\quad & V = \frac{F_{A0}X}{(-r_A)_{\text{exit}}}
 &&\text{rate evaluated at \emph{exit} conditions}\\[6pt]
\text{PFR:}\quad & \frac{dF_A}{dV} = r_A
 &&\Longrightarrow\quad V = F_{A0}\int_0^X \frac{dX}{-r_A}\\[6pt]
\text{PBR:}\quad & \frac{dF_A}{dW} = r_A'
 &&\Longrightarrow\quad W = F_{A0}\int_0^X \frac{dX}{-r_A'}
\end{align*}
\end{formulabox}
\begin{itemize}
\item The structural difference between a \textbf{CSTR} and a \textbf{PFR} is exactly the difference between evaluating $-r_A$ at the exit and integrating it along the path. That single fact drives all of section~2.
\item \textbf{Space time} $\tau = V/v_0$; \textbf{space velocity} $= v_0/V = 1/\tau$.
\end{itemize}

\spsub{1.4 \enspace Worked from the deck}
\begin{itemize}
\item \textbf{Rocket engine} (slide~36). Chamber $75\times60$~cm, exhaust $108$~kg/s.
\end{itemize}
\begin{mthbox}
\begin{align*}

\notetext{The $2{:}1$ ratio of the two rates is the stoichiometry of $2\mathrm{H}_2+\mathrm{O}_2\to2\mathrm{H}_2\mathrm{O}$ --- the first appearance of the relative-rate rule developed in section~3.}

\end{itemize}

\spsection{2 \enspace Conversion and Reactor Sizing}
\notetext{Source: \texttt{Ch2-krd (1).pdf}.}

\spsub{2.1 \enspace The Levenspiel plot}
\begin{itemize}
\item Plot \textbf{$F_{A0}/(-r_A)$ against $X$}. Then the two design equations become areas on that single graph:
\end{itemize}
\begin{formulabox}
\begin{align*}
V_{\text{CSTR}} &= \text{\textbf{rectangle}} = \left.\frac{F_{A0}}{-r_A}\right|_{X}\cdot X
 &&\text{height at the exit } \times \text{ width}\\[5pt]
V_{\text{PFR}} &= \text{\textbf{area under the curve}} = \int_0^X \frac{F_{A0}}{-r_A}\,dX
 &&\text{integral from 0 to } X
\end{align*}
\end{formulabox}
\begin{itemize}
\item For any \emph{positive-order} kinetics $-r_A$ falls as $X$ rises, so $F_{A0}/(-r_A)$ rises and the \textbf{rectangle always exceeds the area under the curve}: a CSTR needs more volume than a PFR for the same conversion. The deck's own numbers in \S1.4 show the gap opening from $0$ to a factor of $100$.
\end{itemize}

\spsub{2.2 \enspace Numerical integration of the PFR integral}
\begin{formulabox}
\begin{align*}
\text{Simpson (3 pt):}\quad & \int \approx \frac{h}{3}\left(f_0 + 4f_1 + f_2\right)\\[3pt]
\text{Simpson (5 pt):}\quad & \int \approx \frac{h}{3}\left(f_0 + 4f_1 + 2f_2 + 4f_3 + f_4\right)\\[3pt]
\text{3/8 rule:}\quad & \int \approx \frac{3h}{8}\left(f_0 + 3f_1 + 3f_2 + f_3\right)
\end{align*}
\end{formulabox}
\notetext{The five-point Simpson rule is the one the deck uses throughout, and it is the one that reproduces the printed answers.}

\spsub{2.3 \enspace Reactors in series}
\begin{formulabox}
\begin{align*}
F_{Ai} &= F_{A0}\left(1 - X_i\right)
 &&\text{flow leaving stage } i\\[3pt]
V_i &= \frac{F_{A0}\left(X_i - X_{i-1}\right)}{(-r_A)_i}
 &&\text{CSTR } i \text{ : rectangle from } X_{i-1} \text{ to } X_i\\[3pt]
V_i &= F_{A0}\int_{X_{i-1}}^{X_i}\frac{dX}{-r_A}
 &&\text{PFR } i \text{ : area over the same interval}
\end{align*}
\end{formulabox}

\spsub{2.4 \enspace Reactors in parallel}
\begin{itemize}
\item \textbf{Rule:} for the total volume to be a minimum the feed must be split so that \textbf{$V/F$ (equivalently $\tau$) is the same in every branch}. Branches running at different $\tau$ waste volume.
\end{itemize}

\spsub{2.5 \enspace Choosing the best arrangement}
\begin{itemize}
\item Work from the \textbf{$1/(-r_A)$ versus $C_A$} curve.
\item \textbf{$n>1$} (rate falls steeply): plug flow first, then a small mixed reactor, then a large mixed reactor.
\item \textbf{$n<1$}: the order is reversed --- mixed flow first.
\item \textbf{Two CSTRs, minimum total volume:} maximise the inscribed rectangle, which gives the tangency condition $-dy/dx = y/x$ on the $1/(-r_A)$ curve.
\end{itemize}

\spsub{2.6 \enspace Worked from the deck --- the sizing table (slides 19--38)}
\notetext{The deck gives $T=500$~K, $P=830$~kPa, $F_{A0}=0.4$~mol/s and this rate table. Note that the third row is $-r_A$ itself, not its reciprocal --- the reciprocal is the last column.}

\item \textbf{Unknown stoichiometry from an MFR} (slide~42). $C_{A0}=0.10$, $C_{B0}=0.01$~mol/L, $V=1$~L, outlet $0.02/0.03/0.04$:

\spsection{3 \enspace Rate Laws and Stoichiometry}
\notetext{Source: \texttt{Ch3-krd.pdf} --- a two-part deck of 63 printed slides: \textbf{Part~1, rate laws} (slides 3--28) and \textbf{Part~2, stoichiometry} (slides 29--60), closing with three numericals (slides 61--63). Slide numbers cited below are the deck's own printed numbers, read off the slide photographs together with the handwritten callouts.}

\spsub{3.1 \enspace Molecularity --- how many particles meet (slide 4)}
\begin{itemize}
\item \textbf{Molecularity} is the number of atoms, ions or molecules involved in a single reaction event. The deck classifies reactions by this count and gives one example of each:
\end{itemize}
\begin{formulabox}
\begin{align*}
\text{unimolecular:}\quad
& {}^{238}\mathrm{U} \longrightarrow {}^{234}\mathrm{Th} + {}^{4}\mathrm{He}
 &&\text{(a single nucleus disintegrates)}\\[3pt]
\text{bimolecular:}\quad
& \mathrm{Br}^{\boldsymbol{\cdot}} + \mathrm{C_2H_6} \longrightarrow \mathrm{HBr} + \mathrm{C_2H_5}^{\boldsymbol{\cdot}}
 &&\text{(a radical meets a molecule)}\\[3pt]
\text{trimolecular:}\quad
& 2\,\mathrm{NO} + \mathrm{Cl_2} \longrightarrow 2\,\mathrm{NOCl}
 &&\text{(three particles in one event)}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{Termolecular events are already rare, and molecularity~4 is the wall:} the deck states in red that \emph{there is no known reaction with molecularity 4} --- four particles cannot arrive with the right energy and geometry at the same instant.
\item \textbf{Why the first example is on this slide at all:} the uranium decay shows that ``reaction'' in kinetics is not limited to molecules in a beaker; the counting rules of this chapter apply to any transformation written $aA+bB\to cC+dD$.
\item \textbf{Net reactions hide the real molecularity} (slide~4): most reaction pathways are a \emph{series of bimolecular steps}. The deck's illustration is the overall trimolecular-looking $2\,\mathrm{NO}+\mathrm{O_2}\to2\,\mathrm{NO_2}$, which in fact proceeds through bimolecular encounters. Molecularity belongs to an \emph{elementary step}; overall order (section~3.3) belongs to the \emph{measured} rate law, and the two need not agree --- the theme of slides 14--17.
\end{itemize}

\spsub{3.2 \enspace Relative rates of reaction (slides 5--7)}
\begin{itemize}
\item Every species in $aA+bB\to cC+dD$ has its own rate of reaction $r_i$, but the stoichiometry locks the ratios. Divide each species' rate by its own stoichiometric coefficient \emph{taken with sign} (negative for reactants, positive for products); all four quotients are equal (slide~5, callout ``Reaction Stoichiometry''):
\end{itemize}
\begin{formulabox}
\begin{align*}
aA + bB &\longrightarrow cC + dD\\[4pt]
\frac{r_A}{-a} &= \frac{r_B}{-b} = \frac{r_C}{c} = \frac{r_D}{d}
 &&\Longrightarrow\quad \boxed{\;r_B = \frac{b}{a}\,r_A\;}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{Sign convention, stated once and used everywhere:} $r_i>0$ means species~$i$ is being \emph{formed}, $r_i<0$ means it is being \emph{consumed}. The rate of \emph{disappearance} of a reactant is $-r_i>0$. Mixing the two readings is the classic way to lose a sign.
\item \textbf{Why this closes Chapter~1:} the rocket-engine problem of section~1.4 balances $-r_{\mathrm{H_2}}$ against $-r_{\mathrm{O_2}}$ exactly this way ($2{:}1$, the coefficients of $2\mathrm{H_2}+\mathrm{O_2}\to2\mathrm{H_2O}$).
\end{itemize}

\spsub{3.2b \enspace The two worked relative-rate problems (slides 5--7)}
\begin{qbox}
\textbf{Example (slides 5 and the sticky note).} For $2\,\mathrm{NO}+\mathrm{O_2}\to2\,\mathrm{NO_2}$ the rule reads
\[
\frac{r_{\mathrm{NO}}}{-2}=\frac{r_{\mathrm{O_2}}}{-1}=\frac{r_{\mathrm{NO_2}}}{2}.
\]
The handwritten note asks: \emph{``If $\mathrm{NO_2}$ formed at $4$~mol/m$^3\cdot$s, ($r_{\mathrm{NO_2}}=+4$~mol/m$^3\cdot$s), what is the rate of formation of NO?''}
\end{qbox}
\begin{ansbox}
$r_{\mathrm{NO}}=\dfrac{-2}{2}(+4)=-4$~mol\,m$^{-3}$s$^{-1}$ and $r_{\mathrm{O_2}}=\dfrac{-1}{2}(+4)=-2$~mol\,m$^{-3}$s$^{-1}$.\newline
The \emph{rate of formation} of NO is $-4$~mol\,m$^{-3}$s$^{-1}$: the minus sign is the answer --- NO is not formed at all, it is consumed at $4$~mol\,m$^{-3}$s$^{-1}$.
\end{ansbox}
\begin{qbox}
\textbf{Exercise (slide 6), solved on slide 7.} The reaction $2A+3B\to5C$ is carried out in a reactor. At a particular point the rate of disappearance of $A$ is $10$~mol/dm$^3\cdot$s. What are the rates of $B$ and $C$?
\end{qbox}
\begin{mthbox}
\begin{align*}
\frac{r_A}{-2} &= \frac{r_B}{-3} = \frac{r_C}{5},
 &\text{given } -r_A = 10 \text{ mol/dm}^3\text{/s, i.e. } r_A = -10;\\[4pt]
r_B &= \frac{-3}{-2}\,(-10) = -15\ \mathrm{mol\,dm^{-3}s^{-1}},
 &\qquad
r_C = \frac{5}{-2}\,(-10) = +25\ \mathrm{mol\,dm^{-3}s^{-1}} .
\end{align*}
\end{mthbox}
\notetext{Read the result physically: $B$ is a reactant, so $r_B<0$ (it disappears at $15$~mol/dm$^3\cdot$s); $C$ is a product, so $r_C>0$ (it appears at $25$). The three rates are in the ratio $2{:}3{:}5$ --- the stoichiometry again.}

\spsub{3.3 \enspace The rate law: what it is and how it is built (slides 8, 12--13)}
\begin{itemize}
\item \textbf{Definition (slide 12, callout bubble):} a \emph{rate law} is a kinetic expression that gives the relationship between reaction rate and concentration. The rate of disappearance depends on \emph{temperature} and \emph{composition}, and the law is built by splitting those two dependencies:
\end{itemize}
\begin{formulabox}
\[
-r_A = \underbrace{\big[k_A(T)\big]}_{\text{temperature}}\;
       \underbrace{\big[f_n(C_A,\,C_B,\dots)\big]}_{\text{composition}}
\]
\end{formulabox}
\begin{itemize}
\item The rate constant $k_A$ carries \emph{all} the temperature dependence (section~3.7); everything about \emph{who} reacts with \emph{whom} sits in the composition function.
\item The subscript $A$ in $k_A$ exists because the numerical value of $k$ depends on \emph{which species' rate} is being written. When the stoichiometric coefficient of every species is~1 --- the deck's example $\mathrm{NaOH}+\mathrm{HCl}\to\mathrm{NaCl}+\mathrm{H_2O}$ (slide~13) --- the subscripts may be dropped:
\[
k = k_{\mathrm{NaOH}} = k_{\mathrm{HCl}} = k_{\mathrm{NaCl}} = k_{\mathrm{H_2O}} .
\]
\end{itemize}

\spsub{3.4 \enspace Power-law kinetics, orders and the units of $k$ (slides 8--10)}
\begin{itemize}
\item The working assumption of this chapter (slide~8): the reaction $\sum_j \nu_j A_j=0$ takes place in a \textbf{closed, isothermal, constant-pressure} system --- a batch at uniform $T$ and $P$. Then the composition function is a \emph{power law}, a product of concentrations raised to exponents:
\end{itemize}
\begin{formulabox}
\begin{align*}
r &= k\,C_1^{q_1}C_2^{q_2}\cdots C_N^{q_N} = k\prod_{j=1}^{N} C_j^{q_j}
 &&\text{(general power law, slide 8)}\\[5pt]
-r_A &= k\,C_A^{\alpha}\,C_B^{\beta}
 &&\text{(two-reactant form)}\\[5pt]
n &= \alpha + \beta
 &&\text{overall order $=$ sum of the partial orders}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{Language of orders (slide 8):} the reaction is $\alpha$ order with respect to reactant~$A$ and $\beta$ order with respect to reactant~$B$; $n=\alpha+\beta$ is the \emph{overall order}. Orders measure how violently the rate responds to each concentration: double $C_A$ and $-r_A$ multiplies by $2^{\alpha}$.
\item \textbf{Law of mass action (slide 10):} the exponents are \emph{not} free parameters at this stage. For the extent-based rate $r$ of $\sum_j\nu_jA_j=0$ the deck writes $q_j$ in terms of the stoichiometric coefficients $\nu_j$ (negative for reactants, positive for products):
\end{itemize}
\begin{formulabox}
\[
r = k\prod_{j=1}^{N} C_j^{q_j},
\qquad
q_j = \tfrac{1}{2}\big(|\nu_j| - \nu_j\big)
\]
\end{formulabox}
\begin{itemize}
\item The combination $\tfrac12(|\nu_j|-\nu_j)$ is a neat switch: it returns $|\nu_j|$ for a reactant ($\nu_j<0$) and $0$ for a product ($\nu_j>0$) --- so the forward power law involves reactant concentrations only. Section~3.5 needs the mirror combination $\tfrac12(|\nu_j|+\nu_j)$, which does the opposite (products only).
\item \textbf{The units of $k$ betray the order (slide 9).} Write any rate law as $-r_A = k\,C_A^{n}$ and demand that both sides carry (concentration)/(time):
\end{itemize}
\begin{formulabox}
\[
k = \frac{(\text{Concentration})^{1-n}}{\text{Time}}
\]
\end{formulabox}
{\small
\begin{tabularx}{\linewidth}{@{}l l X@{}}
\toprule
\textbf{\color{accent}Order} & \textbf{Rate law ($A\to$ products)} & \textbf{Units of $k$ (deck's print)}\\
\midrule
Zero $(n=0)$ & $-r_A = k_A$ & $\{k\} = \mathrm{mol\,dm^{-3}\,s^{-1}}$\\[2pt]
First $(n=1)$ & $-r_A = k_A C_A$ & $\{k\} = \mathrm{s^{-1}}$\\[2pt]
Second $(n=2)$ & $-r_A = k_A C_A^2$ & $\{k\} = \mathrm{dm^{3}\,mol^{-1}\,s^{-1}}$\\[2pt]
Third $(n=3)$ & $-r_A = k_A C_A^3$ & $\{k\} = \left(\mathrm{dm^{3}/mol}\right)^{2}\,\mathrm{s^{-1}}$\\
\bottomrule
\end{tabularx}}
\notetext{Practical use: convert units of concentration first (mol/L, mol/m$^3$, mmol/L \dots), because $k$ changes its number whenever the concentration unit changes. A rate constant quoted in s$^{-1}$ is first order \emph{whatever} the chemistry claims; one quoted in (dm$^3$/mol)$^2$\,s$^{-1}$ is third order.}

\spsub{3.5 \enspace Elementary, non-elementary and elementary-looking laws (slides 14--17)}
\begin{itemize}
\item \textbf{Elementary reaction (slide 14):} a chemical reaction in which one or more of the chemical species react directly to form products in a \emph{single reaction step} and with a \emph{single transition state}. Prototypes: $A\to\text{products}$ (unimolecular) and $A+B\to\text{products}$ (bimolecular).
\item \textbf{Elementary rate law (slide 14):} the reaction \emph{follows the elementary rate law} if the stoichiometric coefficients are \textbf{identical} to the reaction order of each species. Then the law of mass action (section~3.4) is literally true and no experiment can disagree.
\item \textbf{The trap (slide 14):} $2\,\mathrm{NO}+\mathrm{O_2}\to2\,\mathrm{NO_2}$ is flagged on the slide as a \emph{non-elementary} reaction --- its mechanism is a series of bimolecular steps --- \emph{but} its measured law $-r_{\mathrm{NO}} = k_{\mathrm{NO}}\,C_{\mathrm{NO}}^2\,C_{\mathrm{O_2}}$ happens to look exactly like the elementary form (orders $2$ and $1$ = coefficients). Matching coefficients is therefore \emph{not} proof of an elementary mechanism; it is only permission to use the simple law.
\item \textbf{Non-elementary rate laws (slides 15--17)} are those that \emph{do not follow} simple power laws --- fractional orders, denominators, products in numerators. The deck's catalogue:
\end{itemize}
\begin{itemize}
\item[] \textbf{(a) Homogeneous, fractional orders (slide 15).} Gas-phase synthesis of phosgene, $\mathrm{CO}+\mathrm{Cl_2}\to\mathrm{COCl_2}$:
\end{itemize}
\begin{formulabox}
\[
-r_{\mathrm{CO}} = k\,C_{\mathrm{CO}}\,C_{\mathrm{Cl_2}}^{3/2}
\qquad
\text{first order in CO, three-halves in $\mathrm{Cl_2}$, five-halves overall.}
\]
\end{formulabox}
\begin{itemize}
\item[] Half-integer orders cannot come from any single elementary step --- they are the fingerprint of a hidden mechanism.
\item[] \textbf{(b) Heterogeneous, on a catalyst (slides 16--17; the deck prints this example on both).} Gas-solid catalysed \emph{hydrodemethylation of toluene (T)}:
 $\mathrm{C_6H_5CH_3}+\mathrm{H_2}\xrightarrow{\mathrm{cat}}\mathrm{C_6H_6}+\mathrm{CH_4}$, written $T+H_2\to B+M$. The rate of disappearance of toluene \emph{per mass of catalyst} (hence the prime on $r'$):
\end{itemize}
\begin{formulabox}
\[
-r_T' = \frac{k\,P_H\,P_T}{1 + K_B P_B + K_T P_T}
\qquad
\text{$K_B$, $K_T$ = adsorption constants.}
\]
\end{formulabox}
\begin{itemize}
\item[] Two things to notice (slide 16 callout): the law is written \emph{in terms of partial pressures rather than concentrations} --- on a surface the coverage, not the molarity, controls the rate --- and the denominator is the adsorption competition of products and reactants for the same sites (developed fully in section~8). Partial pressures and concentrations are interchangeable through $P_i = C_iRT$ only in ideal gases; the \emph{form} of the law is what matters here.
\item[] \textbf{(c) Cumene dehydrogenation (slide 24, listing C).} Cumene (C) $\to$ benzene (B) $+$ propylene (P) over a catalyst, again per mass of catalyst:
\end{itemize}
\begin{formulabox}
\[
-r_C = \frac{k\left(P_C - \dfrac{P_B P_P}{K}\right)}{1 + K_B P_B + K_C P_C}
\]
\end{formulabox}
\begin{itemize}
\item[] The numerator is \emph{forward $-$ reverse} (the reaction is reversible; section~3.6); the denominator again collects adsorption terms. This single line already combines every ingredient of this chapter and of section~8.

\spsub{3.6 \enspace Reversible rate laws (slides 11, 18--21)}
\item[] \textbf{The thermodynamic straitjacket (slide 18).} For $aA+bB\rightleftharpoons cC+dD$, \emph{all rate laws must reduce to the thermodynamic relationship relating the reacting species concentrations at equilibrium} --- the slide's orange callout names it the Thermodynamic Equilibrium Relationship:
\end{itemize}
\begin{formulabox}
\[
K_C = \frac{C_{Ce}^{\,c}\,C_{De}^{\,d}}{C_{Ae}^{\,a}\,C_{Be}^{\,b}}
\qquad
\text{(concentrations at equilibrium carry the subscript $e$)}
\]
\end{formulabox}
\begin{itemize}
\item[] At equilibrium the net rate must vanish; therefore a reversible rate law is always a \emph{difference of two terms}, forward minus reverse, and the ratio of the two rate constants is tied to $K_C$:
\end{itemize}
\begin{formulabox}
\begin{align*}
r &= k_f \prod_{j=1}^{N} C_j^{q_j} \;-\; k_r \prod_{j=1}^{N} C_j^{q_j^{*}}
 &&\text{(slide 11, general form)}\\[4pt]
q_j &= \tfrac{1}{2}\big(|\nu_j|-\nu_j\big),
 &\qquad
q_j^{*} &= \tfrac{1}{2}\big(|\nu_j|+\nu_j\big)
\end{align*}
\end{formulabox}
\begin{itemize}
\item[] The pair $\tfrac12(|\nu_j|\mp\nu_j)$ splits every species into ``reactant side'' ($q_j$) or ``product side'' ($q_j^*$), so the forward product runs on reactant concentrations and the reverse product on product concentrations. Setting $r=0$ and rearranging gives $k_f/k_r = K_C$ --- exactly the thermodynamic constraint above.
\item[] \textbf{Worked in full: benzene dimerisation (slides 19--21).} The deck's reaction is
 $2\,\mathrm{C_6H_6}\rightleftharpoons\mathrm{C_{12}H_{10}}+\mathrm{H_2}$, written symbolically $2B\rightleftharpoons D+H_2$ (diphenyl $D$). Build the law step by step as the slides do:
\end{itemize}
\begin{mthbox}
\textbf{Step 1 --- the two one-way rates} (slide 19). Forward: benzene disappears, second order;
 reverse: diphenyl and hydrogen recombine, first order in each.
\[
-r_{B,\mathrm{forward}} = k_B C_B^2
\qquad\text{(equivalently } r_{B,\mathrm{forward}} = -k_B C_B^2\text{)}
\]
\[
r_{B,\mathrm{reverse}} = k_{-B}\,C_D\,C_{\mathrm{H_2}}
\]
\textbf{Step 2 --- net formation of benzene} (slide 20):
\[
r_B \equiv r_{B,\mathrm{net}} = r_{B,\mathrm{forward}} + r_{B,\mathrm{reverse}}
= -k_B C_B^2 + k_{-B} C_D C_{\mathrm{H_2}}
\]
\textbf{Step 3 --- multiply by $-1$} to get the rate of \emph{disappearance} of benzene, and factorise the forward constant out:
\[
-r_B = k_B C_B^2 - k_{-B} C_D C_{\mathrm{H_2}}
= k_B\!\left(C_B^2 - \frac{k_{-B}}{k_B}\,C_D C_{\mathrm{H_2}}\right)
\]
\textbf{Step 4 --- replace the constant ratio by $K_C$} (slide 21): since $k_B/k_{-B}=K_C$ (Concentration equilibrium constant),
\[
\boxed{\;-r_B = k_B\!\left(C_B^2 - \frac{C_D\,C_{\mathrm{H_2}}}{K_C}\right)\;}
\]
\end{mthbox}
\notetext{The structure of the boxed law is the reusable part: \emph{(forward term)} $\times$ $\big(1 - (\text{reverse})/(K_C\cdot\text{forward without }k)\big)$. At equilibrium the bracket vanishes, recovering $K_C = C_D C_{\mathrm{H_2}}/C_B^2$ --- which is slide~18's thermodynamic check done on a real example. Large $K_C$ pushes the bracket toward~1 and the reaction behaves irreversibly; chapter~5 uses exactly this limiting behaviour.}

\spsub{3.8 \enspace The rate constant $k(T)$: Arrhenius (slides 25--28)}
\begin{itemize}
\item \textbf{The Arrhenius equation (slide 25, callout bubble):} all the temperature dependence of a rate law hides inside $k_A(T)$:
\end{itemize}
\begin{formulabox}
\[
k_A(T) = A\,e^{-E/RT}
\]
\begin{align*}
A &= \text{pre-exponential factor (frequency factor)}\\
E &= \text{activation energy, J/mol or cal/mol}\\
R &= \text{gas constant} = 8.314\ \mathrm{J\,mol^{-1}K^{-1}} = 1.987\ \mathrm{cal\,mol^{-1}K^{-1}}\\
T &= \text{absolute temperature, K}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{Limits (the $k$-vs-$T$ curve on slide 25):} $T\to\infty$ gives $k\to A$ --- the pre-exponential factor is the ceiling of the rate constant --- and $T\to 0$ gives $k\to0$. The curve is flat at low $T$, steep in the middle, and levels off at $A$; almost every industrial reactor lives in the steep middle.
\item \textbf{Collision reading of the two factors (slides 25--26):}
\end{itemize}
\begin{formulabox}
\begin{align*}
A &: \text{total number of collisions}\\
e^{-E/RT} &: \text{probability that a collision results in a reaction}\\
k &: \text{number of collisions that result in a reaction}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{Activation energy (slide 26)} is a measure of the \emph{minimum energy the reacting molecules must have} in order for the reaction to occur --- the energy required to reach the transition state. The slide's energy diagram: reactants $\mathrm{AB}+\mathrm{C}$ climb over the barrier to the activated complex $\mathrm{A\cdots B\cdots C}$ (transition state at the saddle) and roll down to products $\mathrm{A}+\mathrm{BC}$. The barrier height is $E$; a catalyst works by lowering it (section~8), never by changing the end states.
\item \textbf{Logarithmic form (slide 27)} --- the working form for data analysis:
\end{itemize}
\begin{formulabox}
\[
k_A(T) = A\,e^{-E/RT}
\qquad\Longrightarrow\qquad
\ln k_A = \ln A - \frac{E}{R}\cdot\frac{1}{T}
\]
\end{formulabox}
\begin{itemize}
\item \textbf{The slide's $E$--$k$--$r$ shorthand (slide 27):} at a given temperature the activation energy and the rate constant move in \emph{opposite} directions ($E\uparrow\Rightarrow k\downarrow\Rightarrow r\downarrow$; $E\downarrow\Rightarrow k\uparrow\Rightarrow r\uparrow$). Everything the temperature does to a reactor passes through this one exponential.
\item \textbf{Two-temperature form} --- subtract the logarithmic form written at two temperatures (the form the deck itself uses in the slide-61 pasteurisation problem):
\end{itemize}
\begin{formulabox}
\[
\boxed{\;\ln\frac{k_2}{k_1} = \frac{E}{R}\left(\frac{1}{T_1}-\frac{1}{T_2}\right)\;}
\qquad\Longleftrightarrow\qquad
E = \frac{R\,\ln(t_1/t_2)}{1/T_1 - 1/T_2}\ \ \text{(equal first-order conversions)}
\]
\end{formulabox}
\begin{itemize}
\item \textbf{The Arrhenius plot (slide 28)} is $\ln k$ vs $1/T$: a straight line of slope $-E/R$ and intercept $\ln A$. Two runs give $E$ without knowing $A$; five runs give a regression (section~5). The slide's conclusions, verbatim in substance:
\end{itemize}
\begin{itemize}
\item[] \textbf{1.} Reactions with high activation energies are very temperature-sensitive; reactions with low activation energies are relatively temperature-insensitive.
\item[] \textbf{2.} Any given reaction is much more temperature-sensitive at a low temperature than at a high temperature.
\item[] (Callout bubble:) \emph{the larger the activation energy, the more temperature sensitive $k$ and thus the reaction rate.}
\end{itemize}
\notetext{Point~2 is quantitative: $\dfrac{d\ln k}{dT} = \dfrac{E}{RT^2}$ --- the same $E$ produces a far bigger fractional change in $k$ at small~$T$. This single derivative reappears as the slope of van't Hoff-style plots and as the sensitivity knob in the non-isothermal design of section~7.}

\spsub{3.9 \enspace Part 2 opens: why stoichiometric tables (slides 29--32)}
\begin{itemize}
\item \textbf{Purpose of developing a stoichiometric table (slide 29):} to determine the number of moles of each species \emph{remaining} at a conversion $X$. The rate law needs concentrations; the reactor design equations of Chapter~2 need concentrations as functions of $X$; the table is the machine that manufactures them.
\item \textbf{Components of the table (slide 30):} columns \emph{Species | Initially | Change | Remaining}. The slide's handwritten callout: \emph{``Change refers to moles of species reacted or formed''} --- negative entries for what is consumed, positive for what is made. Rows for every reactant and product, one for inerts (I), one for totals.
\item \textbf{Conversion recalls Chapter~2 (slide 31)} for $aA+bB\to cC+dD$:
\end{itemize}
\begin{formulabox}
\[
X = \frac{N_{A0}-N_A}{N_{A0}}
\qquad\Longrightarrow\qquad
N_A = N_{A0}-N_{A0}X
\qquad\Longrightarrow\qquad
\boxed{\;N_A = N_{A0}(1-X)\;}
\]
\end{formulabox}
\notetext{Callouts on slide 31: the numerator $N_{A0}-N_A$ is the \emph{moles of A reacted}; the boxed $N_A$ is the \emph{moles of A remaining in the reactor at a conversion of X}. Everything else is bookkeeping in the same style.}
\begin{itemize}
\item \textbf{Every other row follows from one ratio} (slide 32): moles reacted or formed are tied to moles of $A$ reacted by the stoichiometry,
\end{itemize}
\begin{formulabox}
\begin{align*}
\text{moles B reacted} &= \frac{\text{moles B reacted}}{\text{moles A reacted}}\;\big\{\text{moles A reacted}\big\} = \frac{b}{a}\,(N_{A0}X)\\[3pt]
\text{moles C formed} &= \frac{c}{a}\,(N_{A0}X),
\qquad
\text{moles D formed} = \frac{d}{a}\,(N_{A0}X)
\end{align*}
\end{formulabox}
\begin{itemize}
\item Subtract the reacted amount from the initial amount (reactants) or add the formed amount (products) and the remaining moles follow (slide 33):
\end{itemize}
\begin{formulabox}
\[
N_B = N_{B0} - \frac{b}{a}N_{A0}X,
\qquad
N_C = N_{C0} + \frac{c}{a}N_{A0}X,
\qquad
N_D = N_{D0} + \frac{d}{a}N_{A0}X
\]
\end{formulabox}

\begin{itemize}
\item \textbf{The total row defines $\delta$ (slides 34--35):} add up the changes and the coefficient of $N_{A0}X$ is the change in the \emph{total} number of moles \emph{per mole of $A$ reacted} --- the slide's red circle and the phrase ``change in the total number of moles per mole of A reacted'':
\end{itemize}
\begin{formulabox}
\[

\qquad
\boxed{\;\delta = \frac{d}{a}+\frac{c}{a}-\frac{b}{a}-1\;}
\]
\end{formulabox}
\begin{itemize}
\item \textbf{Interpretation:} $\delta>0$ (moles created, e.g.\ $A\to 3R$) makes a constant-$T,P$ gas \emph{expand}; $\delta<0$ (moles destroyed, e.g.\ $2A\to R$) makes it \emph{contract}; $\delta = 0$ (equal moles each side) freezes the total. Inerts do not appear in $\delta$ --- they never react.
\item \textbf{Why the table must also carry concentrations (slide 36):} the rate laws waiting downstream are $-r_A = kC_A$, $kC_A^2$, $kC_A^3$ --- functions of \emph{concentration}, not of moles. The next slides divide every row by the volume.
\end{itemize}

\spsub{3.11 \enspace Concentration in terms of conversion --- batch liquids (slides 36--41)}
\notetext{For a liquid-phase batch the volume is constant, $V = V_0$ --- the deck states this explicitly on every liquid slide. Divide the remaining-moles rows by $V$ (slide 37):}

\Theta_i = \frac{N_{i0}}{N_{A0}} = \frac{C_{i0}}{C_{A0}} = \frac{y_{i0}}{y_{A0}}

\end{formulabox}

\item With $\Theta_B = N_{B0}/N_{A0}$ the product $N_{A0}\Theta_B$ simply \emph{is} $N_{B0}$ (slide 40 annotation), and every concentration becomes $C_{A0}\times(\text{a function of }X)$ --- which is the form the rate laws want. The complete liquid-batch set (slides 40--41):

\spsub{3.15 \enspace Concentration in terms of conversion --- the four cases (slides 52--60)}
\notetext{Part~2's finale is a single master formula assembled case by case. Two questions decide everything: \emph{does the volume change?} (liquids: no; gases: yes) and \emph{is it batch ($N$) or flow ($F$)?}}

\textbf{\color{accent}Case 1 --- liquid, batch (slide 52):} $V = V_0$ (boxed on the slide), so division by $V_0$ gives at once
\[
C_A = C_{A0}(1-X),\quad
C_B = C_{A0}\!\left[\Theta_B-\tfrac{b}{a}X\right],\quad
C_C = C_{A0}\!\left[\Theta_C+\tfrac{c}{a}X\right],\quad
C_D = C_{A0}\!\left[\Theta_D+\tfrac{d}{a}X\right].
\]

\textbf{\color{accent}Case 2 --- liquid, flow (slide 53):} liquids are incompressible, $v = v_0$ (boxed), so $C_i = F_i/v = F_i/v_0$ and \emph{the same four formulas} hold with $N\to F$ already built in:
\[
C_A = \frac{F_{A0}(1-X)}{v} = C_{A0}(1-X),\quad
C_B = C_{A0}\!\left[\Theta_B-\tfrac{b}{a}X\right],\ \dots
\]

\textbf{\color{accent}Case 3 --- gas, batch (slides 54--56):} now $V$ must come from the equation of state. The deck writes the real-gas law at any time $t$ and at the initial condition:
\begin{align*}
P V &= Z\,N_T\,R\,T \tag*{(1)}\\
P_0 V_0 &= Z_0\,N_{T0}\,R\,T_0 \tag*{(2)}
\end{align*}
with the slide's bubble: $T$ in K, $P$ = total pressure in atm ($1\ \mathrm{atm} = 101.3$~kPa), $Z$ = compressibility factor, $R = 0.08206\ \mathrm{dm^3\,atm\,mol^{-1}K^{-1}}$ (the pressure-units gas constant). Divide (1) by (2) and solve for $V$ (slide 55):
\[
V = V_0\left(\frac{P_0}{P}\right)\frac{T}{T_0}\left(\frac{Z}{Z_0}\right)\frac{N_T}{N_{T0}} \tag*{(3)}
\]
If $Z$ does not change significantly during the reaction, $Z_0\approx Z$ (slide 55 annotation) and the compressibility ratio cancels. It remains to convert the mole ratio --- the job of $\varepsilon$ (slide 56):
\begin{formulabox}
\[
\frac{N_T}{N_{T0}} = 1 + \delta\,y_{A0}X = 1 + \varepsilon X,
\qquad
\boxed{\;\varepsilon = y_{A0}\,\delta = \left(\frac{d}{a}+\frac{c}{a}-\frac{b}{a}-1\right)\frac{N_{A0}}{N_{T0}}\;}
\]
\[
V = V_0\left(\frac{P_0}{P}\right)(1+\varepsilon X)\,\frac{T}{T_0} \tag*{(5, constant $Z$)}
\]
\end{formulabox}
\begin{itemize}
\item \textbf{The equivalent readings of $\varepsilon$} (slide 56 gives all of them; they are useful checks on any calculation):
\end{itemize}
\begin{formulabox}
\[
\varepsilon = \frac{N_T - N_{T0}}{N_{T0}\,X}
\]
and, at complete conversion ($X=1$, irreversible, $N_T \to N_{Tf}$):
\[
\varepsilon = \frac{N_{Tf}-N_{T0}}{N_{T0}} = \frac{V_{X_A=1}-V_{X_A=0}}{V_{X_A=0}}
\]
\end{formulabox}
\begin{itemize}
\item[] \textbf{Words (slide 56):} $\varepsilon$ is the \emph{fractional change in volume of the system between no conversion and complete conversion of reactant~$A$}. The annotation stresses that $N_T/N_{T0} = 1+\varepsilon X$ \emph{applies for both batch and flow systems} --- it is a statement about the \emph{gas}, not about the vessel.
\item[] \textbf{The three regimes:} $\varepsilon = 0$ (no mole change, or so diluted that $y_{A0}\to0$) freezes the volume and all formulas collapse to the liquid ones; $\varepsilon>0$ expands the gas, diluting everything faster than $(1-X)$; $\varepsilon<0$ contracts it. Setting $\varepsilon=0$ is the single most common legitimate simplification --- the deck's own gas examples state it explicitly before using it.

\textbf{\color{accent}Case 4 --- gas, flow (slides 57--60):} identical logic on molar flow rates and the volumetric flow $v$. At any point $C_T = F_T/v = P/(ZRT)$ \textit{(1)}, at the entrance $C_{T0} = F_{T0}/v_0 = P_0/(Z_0RT_0)$ \textit{(2)}; dividing \textit{(1)} by \textit{(2)} and using the flow-table total row $F_T = F_{T0}+F_{A0}\delta X$ (slide 58 annotation):
\[
v = v_0\,\frac{F_T}{F_{T0}}\,\frac{P_0}{P}\,\frac{T}{T_0}
= v_0\!\left(1+\frac{F_{A0}}{F_{T0}}\delta X\right)\frac{P_0}{P}\frac{T}{T_0}
= v_0\,(1+\varepsilon X)\,\frac{P_0}{P}\,\frac{T}{T_0} \tag*{(4)}
\]
(the same $\varepsilon = y_{A0}\delta$; $F_{A0}/F_{T0} = y_{A0}$). Finally substitute $v$ from (4) and $F_j = F_{A0}(\Theta_j + \nu_j X)$ \textit{(5)} --- the slide annotates $\nu_j$ as the stoichiometric coefficient of species $j$ with sign ($d/a$, $c/a$, $-b/a$, $-a$) --- into $C_j = F_j/v$ (slides 59--60):
\end{itemize}
\begin{formulabox}
\[
\boxed{\;C_j = C_{A0}\,\frac{\Theta_j + \nu_j X}{1+\varepsilon X}\cdot\frac{P}{P_0}\cdot\frac{T_0}{T}\;}
\]
\begin{align*}
C_A &= C_{A0}\,\frac{1-X}{1+\varepsilon X}\cdot\frac{T_0}{T}\cdot\frac{P}{P_0},\\[5pt]
C_B &= C_{A0}\,\frac{\Theta_B - (b/a)X}{1+\varepsilon X}\cdot\frac{T_0}{T}\cdot\frac{P}{P_0},\\[5pt]
C_C &= C_{A0}\,\frac{\Theta_C + (c/a)X}{1+\varepsilon X}\cdot\frac{T_0}{T}\cdot\frac{P}{P_0},\\[5pt]
C_D &= C_{A0}\,\frac{\Theta_D + (d/a)X}{1+\varepsilon X}\cdot\frac{T_0}{T}\cdot\frac{P}{P_0},\\[5pt]
C_I &= \frac{C_{A0}\,\Theta_I}{1+\varepsilon X}\cdot\frac{T_0}{T}\cdot\frac{P}{P_0}
\end{align*}
\end{formulabox}
\notetext{The three ratios are each easy to misplace; the sanity check is limits. At $X=0$ every formula must return $C_{A0}\Theta_j$; at constant $T,P$ and $\varepsilon=0$ they must reduce to the liquid forms; and $P/P_0$ appears \emph{first power up} because compression ($P>P_0$) concentrates the gas. Note $\nu_A = -a/a = -1$ per the slide's sign list, which is how $C_A$ acquires $(1-X)$. Sections 4 and 7 plug these $C_j(X)$ expressions straight into the design equations.}

\spsub{3.16 \enspace The three closing numericals (slides 61--63)}
\begin{qbox}
\textbf{Pasteurisation (slide 61).} \emph{``Milk is pasteurized if it is heated to $63\,^{\circ}$C for 30~min, but if it is heated to $74\,^{\circ}$C it only needs 15~s for the same result. Find the activation energy of this sterilization process.''}
\end{qbox}
\begin{mthbox}
Sterilisation is first-order killing: the same result means the same $k\,t$. Equating $k_1t_1 = k_2t_2$ with Arrhenius $k$'s and cancelling $A$ gives the two-temperature form of section~3.8:
\[
E = \frac{R\,\ln(t_1/t_2)}{1/T_1 - 1/T_2}
= \frac{8.314\,\ln(1800/15)}{1/336.15 - 1/347.15}
= \frac{8.314\times4.7875}{9.43\times10^{-5}}
\qquad\Longrightarrow\qquad
\boxed{\;E \approx 422\ \mathrm{kJ/mol}\;}
\]
(Two $k$'s, two unknowns $E$ and $A$ --- but $A$ cancels, which is why a two-point sterilisation schedule determines $E$ outright.)
\end{mthbox}
\notetext{A note on reading the slide: an earlier pass at this photograph misread the $63\,^{\circ}$C as $80\,^{\circ}$C and concluded the printed pairing must be reversed. The photograph prints \textbf{$63\,^{\circ}$C for 30 min} and \textbf{$74\,^{\circ}$C for 15 s} --- a consistent hot-brief/cool-long pair --- and the calculation above follows the print. The enormous $E$ ($422$~kJ/mol against a typical chemical $40$--$200$) is exactly why pasteurisation is so temperature-sensitive: one extra degree buys a large factor in $k$ (section~3.8, point~2).}

\textbf{Method (worked fully as Q13 in the companion document):} for $2A\to R$ with pure $A$, $\varepsilon = y_{A0}\delta = -\tfrac12$; solve $C_{Af} = C_{A0}\dfrac{1-X}{1+\varepsilon X}$ for $X$, then the MFR balance $-r_A = \dfrac{F_{A0}X}{V} = \dfrac{v_0 C_{A0} X}{V}$ gives one $(C_A,\,-r_A)$ pair per run:
\[
X = 0.2502,\ 0.4996,\ 0.6667,\ 0.7995;
\qquad
-r_A = 2502.2,\ 1498.9,\ 800.0,\ 399.8\ \mathrm{mol\,L^{-1}h^{-1}}
\]
A log--log fit gives $n = 1.96 \approx 2$, and dividing each run by $C_A^2$:
\[
\boxed{\;-r_A \approx 0.34\,C_A^2\;},\qquad
k = 0.3407,\ 0.3369,\ 0.3200,\ 0.3583\ \mathrm{L\,mmol^{-1}h^{-1}}
\]
\end{mthbox}
\notetext{The scatter in $k$ across the four runs ($0.320$ to $0.358$, about $\pm6\,\%$) is the practical test that the assumed order is right: a wrong order makes $k$ drift systematically instead of scattering (section~5 builds this test deliberately). The same numerical reappears in \texttt{Ch5\_-\_krd.pdf} (slide 77) and is solved once, as Q13.}

\begin{qbox}
\textbf{Phosphine cracking in a PFR (slide 63).} \emph{``The homogeneous gas decomposition of phosphine $4\,\mathrm{PH_3}(g)\to\mathrm{P_4}(g)+6\,\mathrm{H_2}(g)$ proceeds at $649\,^{\circ}$C with the first-order rate $-r_{\mathrm{PH_3}} = \left(\tfrac{10}{\mathrm{hr}}\right)C_{\mathrm{PH_3}}$. What size of plug flow reactor operating at $649\,^{\circ}$C and $460$~kPa can produce $80\,\%$ conversion of a feed consisting of 40~mol of pure phosphine per hour?''} The slide's sketch labels the coil $4A\to R+6S$ and asks for $V$.
\end{qbox}
\begin{mthbox}
\textbf{Step 1 --- inlet concentration} at the \emph{reactor} state (ideal gas):
\[
C_{A0} = \frac{P}{RT} = \frac{460\,000}{8.314\times922.15} = 60.0\ \mathrm{mol/m^3}.
\]
\textbf{Step 2 --- expansion factor} from the mole count as written, $4\to(1+6)=7$ moles per 4 reacted (\emph{P$_4$ is one mole} of product molecules):
\[
\delta = \frac{1}{4}+\frac{6}{4}-0-1 = 0.75,
\qquad
\varepsilon = y_{A0}\delta = 0.75 \quad(\text{pure }A).
\]
\textbf{Step 3 --- PFR design equation, first order} with $C_A = C_{A0}(1-X)/(1+\varepsilon X)$:
\[
V = \frac{F_{A0}}{k C_{A0}}\int_0^{X}\frac{1+\varepsilon X}{1-X}\,dX
= \frac{F_{A0}}{k C_{A0}}\left[(1+\varepsilon)\ln\frac{1}{1-X} - \varepsilon X\right].
\]
\textbf{Step 4 --- numbers:} $\dfrac{F_{A0}}{kC_{A0}} = \dfrac{40}{10\times60} = 0.0667$~m$^3$; the bracket at $X=0.8$: $1.75\ln 5 - 0.75(0.8) = 2.817 - 0.6 = 2.217$:
\[
\boxed{\;V = 0.0667\times2.217 = 0.148\ \mathrm{m^3}\ (\approx 148\ \mathrm{L})\;}
\]
\end{mthbox}
\notetext{Watch the mole count: $\mathrm{P_4}$ is \emph{one} mole of molecules (four atoms), so the products are $1+6 = 7$ moles and $\varepsilon = 0.75$. Counting atoms instead (``$4\to10$'') inflates $\varepsilon$ to $1.5$ and the volume to $0.188$~m$^3$ --- the slip corrected here. The other classic slip is forgetting $\varepsilon$ entirely ($V = 0.107$~m$^3$, far too small) or evaluating $C_{A0}$ at STP instead of at $649\,^{\circ}$C$/$~kPa. Worked step by step as Q14 in the companion document.}

\spsub{4.1 \enspace Conversion and the rate written in $X$ (slides~5--6)}
\begin{formulabox}
\[
X = \frac{F_{A0}-F_A}{F_{A0}}\ \ (\text{flow}),\qquad
X = \frac{N_{A0}-N_A}{N_{A0}}\ \ (\text{batch})
\qquad\Longrightarrow\qquad F_A = F_{A0}(1-X)\ \text{or}\ N_A = N_{A0}(1-X)
\]
\[
-r_A = k\,C_{A0}^{\,n}\,\frac{(1-X)^{n}}{(1+\varepsilon X)^{n}}
\qquad\text{(power law combined with $C_A = C_{A0}(1-X)/(1+\varepsilon X)$, constant $T$, $P$)}
\]
\end{formulabox}
\begin{mthbox}
\textbf{Where the $X$-form of the rate comes from.} The rate is always evaluated at the \emph{local}
conditions. For an ideal-gas single reaction the flow table (\S3.13--3.15) gives
$F_A = F_{A0}(1-X)$ and $v = v_0(1+\varepsilon X)$ at constant $T,P$, so
\[
C_A = \frac{F_A}{v} = \frac{F_{A0}(1-X)}{v_0(1+\varepsilon X)} = C_{A0}\,\frac{1-X}{1+\varepsilon X}.
\]
Substituting into the power law $-r_A = kC_A^{\,n}$ gives the boxed form. If $T$ or $P$ vary, append the
master ratios of \S3.15: $C_A = C_{A0}\dfrac{1-X}{1+\varepsilon X}\cdot\dfrac{T_0}{T}\cdot\dfrac{P}{P_0}$.
\end{mthbox}
\begin{daybox}
\textbf{Express the rate in terms of $X$ first; only then substitute $C_A(X)$.} Slides~9--11 draw this as a
loop (rate law $\to$ $C_i$ in $X$ $\to$ design equation $\to$ solve) and slide~12 packages it as the
\textbf{five steps, in this order, every time:}
\begin{enumerate}
\item \textbf{Mole balance} --- pick the reactor, write its design equation (section~1).
\item \textbf{Rate law} --- from the data or the mechanism (section~3).
\item \textbf{Stoichiometry} --- express every $C_i$ in terms of $X$ (the tables of \S3.4).
\item \textbf{Combine} --- substitute 2 and 3 into 1 to get one equation in $X$.
\item \textbf{Evaluate} --- integrate, or solve, and size the reactor.
\end{enumerate}
\end{daybox}
\notetext{Slide~6 is a hand-drawn \textbf{Levenspiel plot}: $1/(-r_A)$ against $X$ with a rectangle sketched to
some target $X$. Its \emph{area} (times $F_{A0}$) is the PFR volume $V = F_{A0}\int dX/(-r_A)$; its \emph{height}
at the exit is the CSTR integrand $V = F_{A0}X/(-r_A)\big|_{\text{exit}}$. Rectangle versus area is the entire
PFR-versus-CSTR comparison in one picture (chapter~2). A final deck caution (slide~7): ``elementary'' is what
licenses writing the rate law from the stoichiometry (mass-action); if a reaction is not elementary, the
exponents come from data only --- and the Arrhenius plot $\ln k$ versus $1/T$ is a straight line of slope $-E/R$
(section~3.8).}

\spsub{4.2 \enspace Closed-form conversions and the Damk\"ohler number (slides~13--16, 25--28)}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\toprule
\textbf{Batch, 1st order} & $t = \dfrac{1}{k}\ln\dfrac{1}{1-X}$\qquad ($X = 1-e^{-kt}$)\\[6pt]
\textbf{Batch, 2nd order} & $t = \dfrac{1}{kC_{A0}}\cdot\dfrac{X}{1-X}$\qquad ($X = \dfrac{kC_{A0}t}{1+kC_{A0}t}$)\\[6pt]
\textbf{CSTR, 1st order} & $X = \dfrac{k\tau}{1+k\tau}$\\[6pt]
\textbf{CSTR, 2nd order} & $X = \dfrac{(1+2Da)-\sqrt{1+4Da}}{2Da}$,\quad $Da = k\tau C_{A0}$\\
\bottomrule
\end{tabularx}
\begin{mthbox}
\textbf{Derivations (two lines each).} Batch, constant volume: $t = C_{A0}\int_0^X dX/(-r_A)$ with
$-r_A = kC_{A0}^{\,n}(1-X)^n$ gives $\int_0^X (1-X)^{-n}dX$: the logarithm for $n=1$ and
$\frac{1}{n-1}\big[(1-X)^{1-n}-1\big]$ otherwise --- the two boxed rows (slide~15's table also lists the
$n$th-order result and the times for \emph{half} life). CSTR: the balance $C_{A0}X = \tau k C_{A0}^{\,n}(1-X)^n$
is \emph{algebraic}: for $n=1$, $X(1+k\tau)=k\tau$; for $n=2$, $Da\,X^2-(1+2Da)X+Da=0$, whose physical root
(the one with $X\to0$ as $Da\to0$) is the minus branch of the quadratic formula (slide~28, ``in terms of Da''---
the deck's own typo).
\end{mthbox}
\begin{formulabox}
\begin{align*}
Da_1 &= k\tau
 &&\text{Damk\"ohler number, first order}\\[3pt]
Da \le 0.1 &\;\Longrightarrow\; X < 0.1
 &&\text{reaction-limited: little conversion}\\[3pt]
Da \ge 10 &\;\Longrightarrow\; X > 0.9
 &&\text{near-complete conversion}
\end{align*}
\end{formulabox}
\notetext{$Da$ is the ratio of the reaction rate to the convective transport rate. It is the quickest way to see, before any integration, whether a proposed reactor is remotely big enough.}

\spsub{4.3 \enspace Reactors in series and in parallel (slides~25--28)}
\begin{formulabox}
\begin{align*}
\text{$N$ equal CSTRs in series, 1st order:}\quad & X = 1 - \frac{1}{(1+k\tau_i)^{N}}\\[4pt]
\text{Parallel CSTRs:}\quad & \text{behave as one large CSTR of the same total volume}
\end{align*}
\end{formulabox}
\begin{mthbox}
\textbf{Series derivation (slide~25).} One CSTR at steady state:
$C_{\text{in}} - C_{\text{out}} = \tau k\,C_{\text{out}} \Rightarrow C_{\text{out}} = C_{\text{in}}/(1+\tau k)$.
Each tank multiplies the same factor, so after $N$ tanks $C_N = C_0/(1+\tau k)^N$ and $X = 1-C_N/C_0$.
Slide~26 plots $X$ versus $N$ at $Da = 0.1,\,0.5,\,1$: the gains are strongly diminishing, and the step from
one tank to two is the biggest single improvement at fixed total volume. \textbf{Parallel:} $m$ identical CSTRs
sharing the feed have $\tau_i = V_i/v_0 = V_{\text{tot}}/(m v_0)$ each, and each solves
$C_{\text{out}} = C_{\text{in}}/(1+k\tau_i)$ with the \emph{same} inlet --- identical to one CSTR of volume
$V_{\text{tot}}$ (slide~27). Splitting changes the hydraulics, not the conversion.
\end{mthbox}
\notetext{Compare with \S2.6: there, splitting the duty into two CSTRs saved $37\,\%$ of the volume. This is the same result in closed form, and it is why industrial trains are staged rather than single-vessel.}

\spsub{4.4 \enspace Variable volume, gas phase (slides~38--46)}
\begin{formulabox}
\[
V = \int_0^X \frac{F_{A0}\,dX}{-r_A}\ \ (\text{PFR}),\qquad
W = \int_0^X \frac{F_{A0}\,dX}{-r_A'}\ \ (\text{PBR, per catalyst mass})
\]
\begin{align*}
\text{Liquid ($\varepsilon = 0$):}\quad
V &= \frac{v_0}{k\,C_{A0}^{\,n-1}}\int_0^X\frac{dX}{(1-X)^{n}}
&&\text{1st: } \tfrac{v_0}{k}\ln\tfrac{1}{1-X};\ 
\text{2nd: } \tfrac{v_0}{kC_{A0}}\tfrac{X}{1-X}\\[6pt]
V_{\text{PFR}} &= \frac{v_0}{kC_{A0}}\left[
 2\varepsilon(1+\varepsilon)\ln(1-X) + \varepsilon^{2}X
 + \frac{(1+\varepsilon)^{2}X}{1-X}\right]
 &&\text{2nd order, gas (slide~42)}
\end{align*}
\end{formulabox}
\begin{mthbox}
\textbf{Derivation of the gas 2nd-order result.} With $C_A = C_{A0}(1-X)/(1+\varepsilon X)$ and
$F_{A0} = v_0C_{A0}$,
\[
V = \frac{v_0}{kC_{A0}}\int_0^X \left(\frac{1+\varepsilon X}{1-X}\right)^{2} dX ,
\]
and the decomposition $\left(\frac{1+\varepsilon X}{1-X}\right)^2 =
\frac{(1+\varepsilon)^2}{(1-X)^2} - \frac{2\varepsilon(1+\varepsilon)}{1-X} + \varepsilon^2$ integrates term
by term to the red box ($\int -\,\frac{2\varepsilon(1+\varepsilon)}{1-X}\,dX = 2\varepsilon(1+\varepsilon)\ln(1-X)$). Check the limit $\varepsilon\to0$: the bracket collapses to $X/(1-X)$, the liquid form.
Slide~43 graphs the $\varepsilon$-dependence for $\varepsilon = -0.5,\ 0,\ 1,\ 2$: expansion ($\varepsilon>0$)
always inflates the required volume; contraction helps.
\end{mthbox}
\notetext{The first-order gas integral used in \S4.10 below is the same procedure one order down:
$\int_0^X\frac{1+\varepsilon X}{1-X}dX = (1+\varepsilon)\ln\frac{1}{1-X} - \varepsilon X$, which for
$\varepsilon = 1$ is $2\ln\frac{1}{1-X} - X$ (the form in the ethane-cracking solution, slide~44).
Slides~45--46 restate the master relations $C_A = C_{A0}\frac{1-X}{1+\varepsilon X}\frac{T_0}{T}\frac{P}{P_0}$
and $v = v_0(1+\varepsilon X)\frac{T}{T_0}\frac{P_0}{P}$ --- nothing new, but they are what makes the
pressure-drop problem of \S4.5--4.7 a \emph{coupled} system.}

\spsub{4.5 \enspace Pressure drop in packed beds: Ergun, $\beta_0$, and $\alpha$ (slides~47--51)}
\begin{formulabox}
\begin{align*}
\frac{dX}{dW} &= \frac{k\,C_{A0}^{2}}{F_{A0}}
 \left(\frac{1-X}{1+\varepsilon X}\right)^{2}\left(\frac{P}{P_0}\right)^{2}
 &&\text{PBR, 2nd order with drop (slide~47)}\\[6pt]
\frac{dP}{dz} &= -\frac{G(1-\phi)}{\rho_g g_c D_p \phi^{3}}
 \left[\frac{150(1-\phi)\mu}{D_p} + 1.75\,G\right]
 &&\text{Ergun (slide~48)}\\[6pt]
\beta_0 &= \frac{G(1-\phi)}{\rho_0 D_p \phi^{3}}
 \left[\frac{150(1-\phi)\mu}{D_p} + 1.75\,G\right]
 &&\text{(slide~49)}\\[6pt]
W &= z\,A_c\,\rho_b = z\,A_c\,(1-\phi)\rho_c
 &&\text{catalyst mass (slide~50)}\\[6pt]
\frac{dP}{dW} &= -\frac{\beta_0}{A_c(1-\phi)\rho_c}\cdot\frac{P_0}{P}\cdot\frac{T}{T_0}\cdot\frac{F_T}{F_{T0}}
\\[6pt]
\alpha &\equiv \frac{2\beta_0}{A_c(1-\phi)\rho_c}\cdot\frac{1}{P_0}
 &&\text{the lumped drop parameter}
\end{align*}
\end{formulabox}
\begin{mthbox}
\textbf{From $dP/dW$ to the reduced-pressure law.} With $p = P/P_0$ and the slide's $\alpha$,
\[
\frac{dp}{dW} = \frac{1}{P_0}\frac{dP}{dW}
 = -\frac{\alpha}{2p}\cdot\frac{T}{T_0}\cdot\frac{F_T}{F_{T0}}
 \qquad\xrightarrow{\ \text{single reaction: } F_T/F_{T0} = 1+\varepsilon X\ }\qquad
\frac{dp}{dW} = -\frac{\alpha}{2p}\,(1+\varepsilon X)\quad(\text{constant }T).
\]
For $\delta = 0$ (hence $\varepsilon = 0$) this \emph{decouples} and integrates in closed form:
$2p\,dp = -\alpha\,dW \Rightarrow p = (1-\alpha W)^{1/2}$ --- the formula of \S4.6.
\end{mthbox}
\begin{itemize}
\item \textbf{$\phi$} is the bed porosity (void fraction), $(1-\phi)$ the solid fraction, $G = \dot m/A_c$ the
superficial mass flux, $D_p$ the particle diameter, $\rho_c$ the \emph{solid} catalyst density and
$\rho_b = (1-\phi)\rho_c$ the \emph{bulk} density. $\beta_0$ is the \textbf{Ergun equation} in the form the deck
uses (viscous $+$ inertial terms). The slide's full $dP/dz$ form multiplies $\beta_0$ by the property corrections
$\frac{P_0}{P}\frac{T}{T_0}\frac{F_T}{F_{T0}}$, and the English-unit form carries
$g_c = 32.174\ \mathrm{lb_m\,ft/(lb_f\,s^2)}$ (force--mass bridge).
\item The grouping $A_c(1-\phi)\rho_c$ in $\alpha$ is purely the conversion from bed \emph{length} to catalyst
\emph{mass}: $zA_c$ is the bed volume, $(1-\phi)$ the solid volume fraction, $\rho_c$ the solid density.
Everything packing- and flow-specific sits in $\beta_0$; everything geometric or material sits outside.
\item Because $C_A$ carries a factor $p$ (and for order $n$, $p^{\,n}$), pressure drop \emph{slows the reaction},
so the conversion achieved in a given catalyst mass is lower than the no-drop calculation predicts. The two
equations ($X$ and $p$) must be solved \textbf{coupled}, not sequentially --- except at $\delta = 0$, where the
$p$-equation runs ahead on its own (\S4.6).
\end{itemize}

\spsub{4.6 \enspace PBR with $\delta = 0$: closed forms for $X$ and $p$ (slides~52--58)}
\begin{formulabox}
\[
p = \left(1-\alpha W\right)^{1/2},\qquad
\frac{X}{1-X} = G^*\left(W - \frac{\alpha W^2}{2}\right),\qquad
G^* \equiv \frac{k\,C_{A0}^{2}}{F_{A0}}\quad(\text{2nd order})
\]
\[
\text{no drop at all}:\qquad \frac{X}{1-X} = G^*\,W .
\]
\end{formulabox}
\begin{mthbox}
\textbf{Derivation (the slide~53--57 sequence).} At $\varepsilon = 0$ the rate is
$-r_A = kC_{A0}^2(1-X)^2\,p^2$, so $\dfrac{dX}{dW} = G^*(1-X)^2 p^2$ while $\dfrac{dp}{dW} = -\dfrac{\alpha}{2p}$
contains no $X$ at all. Integrate the $p$-equation first ($p^2 = 1-\alpha W$), substitute, separate:
\[
\frac{dX}{(1-X)^{2}} = G^*(1-\alpha W)\,dW
\qquad\Longrightarrow\qquad
\frac{X}{1-X} = G^*\Big(W - \frac{\alpha}{2}W^{2}\Big).
\]
The maximum useful catalyst mass is $W_{\max} = 1/\alpha$ (where $p\to0$ and the rate dies) --- the drag
penalty grows quadratically in $W$ while the benefit grows linearly.
\end{mthbox}
\notetext{Q18, the deck's board problem (slides~55--58): PBR for $A+B\to2C$, isothermal, $W = 100$~kg,
$\alpha = 0.0099\ \mathrm{kg^{-1}}$. The slides quote $k_A = 1.5\ \mathrm{dm^6\,mol^{-1}\,kg^{-1}\,min^{-1}}$ and
the results $X = 0.6$ (with drop) and $X = 0.75$ ($\alpha = 0$). The single lumped group that reproduces both is
$G^* = k C_{A0}^2/F_{A0} = 0.03\ \mathrm{kg^{-1}}$ (the value the deck's own numbers back-solve to; the individual
$k_A$, $C_{A0}$, $F_{A0}$ are not separately printed):
\[
G^*\Big(W-\frac{\alpha W^2}{2}\Big) = 0.03\,(100-49.5) = 1.515
\ \Rightarrow\ X = \frac{1.515}{2.515} = 0.602\ (\text{deck: } 0.6),
\qquad
G^*W = 3\ \Rightarrow\ X = \frac{3}{4} = 0.75\ \text{exactly}.
\]
Ignoring pressure drop overstates conversion by $0.15$, a quarter of the achieved value. This is the number to
quote when justifying a coupled solution.}

\spsub{4.7 \enspace The two-parameter $(\alpha,\beta)$ system and pipe friction (slides~59--63, 88--91)}
\begin{formulabox}
\[
\frac{dx}{dz} = \beta\,\frac{(1-x)^{n}P^{n}}{(1+\varepsilon x)^{n}}\,,
\qquad
\frac{dP}{dz} = -\frac{\alpha}{P}\,,
\qquad\text{i.e.}\qquad
P^{2} = P_{\text{in}}^{2} - 2\alpha z
\]
\[
\beta = \frac{k\,A\,F_{A,\text{in}}^{\,n-1}}{P_{\text{in}}^{\,n}\,Q_{\text{in}}^{\,n}}\,,
\qquad
\alpha_{\text{pipe}} = \frac{2 f_F\,\rho_{\text{in}} u_{\text{in}}^{2}\,P_{\text{in}}}{D}
 = \frac{32\,f_F\,\rho_{\text{in}} Q_{\text{in}}^{2}\,P_{\text{in}}}{\pi^{2}D^{5}}
\qquad\text{(isothermal Fanning form, slides~89--91)}
\]
\end{formulabox}
\begin{mthbox}
\textbf{Derivation of $\beta$ (slide~91's box).} Isothermal ideal gas in terms of the volumetric inlet
$Q$: $C_A = \dfrac{F_A}{Q} = \dfrac{F_{A,\text{in}}}{Q_{\text{in}}}\cdot\dfrac{P}{P_{\text{in}}}
\cdot\dfrac{1-x}{1+\varepsilon x}$. With $-r_A = kC_A^{\,n}$ and $F_{A,\text{in}}\,dx/dz = A(-r_A)$,
\[
\frac{dx}{dz} = \frac{kA\,F_{A,\text{in}}^{\,n-1}}{Q_{\text{in}}^{\,n}}
 \cdot\frac{(1-x)^{n}P^{n}}{P_{\text{in}}^{\,n}(1+\varepsilon x)^{n}},
\]
which is the slide's $\beta$ form (for $n = 1$: $\beta = kA/(P_{\text{in}}Q_{\text{in}})$).\\[4pt]
\textbf{Derivation of $\alpha_{\text{pipe}}$.} Fanning friction: $\dfrac{dP}{dz} = -\dfrac{2f_F\rho u^{2}}{D}$.
At constant mass flux $\rho u^{2} = \rho_{\text{in}}u_{\text{in}}^{2}\,P/P_{\text{in}}$, so
$P\,dP = -\big[2f_F\rho_{\text{in}}u_{\text{in}}^{2}P_{\text{in}}/D\big]dz$ --- exactly the $P^{2}$-law above
with $\alpha$ as boxed. (The deck's slide~90 collects the same objects in the Winterton pipe form and the Ergun
$\beta_0$ form of \S4.5; use whichever matches the data given.)
\end{mthbox}
\begin{itemize}
\item \textbf{Series} beds share the same mass flow: their $\alpha$'s add in the $P^{2}$-law and the exit
condition of one bed is the inlet of the next. \textbf{Parallel} channels split $Q$, so each sees smaller $G$
and much less drop --- but the deck's slides~74--75 conclude that shuffling beds between series and parallel
under $\Delta P$ does not buy conversion: only total catalyst and total drop matter. Pumps and blowers
(slides~61--63) are then sized from the integrated $\Delta P$.
\item \textbf{When does $\Delta P$ matter?} The deck's rule of thumb (slides~68--73): once the minor losses
(fittings, valves, entrance/exit) are already a noticeable fraction (the slide uses $\sim35\,\%$) of the pressure
budget. Physically: an $n$th-order gas rate carries $(P/P_0)^{n}$ --- halve the pressure and an $n=2$ rate drops
by four. If $\alpha W \ll 1$ (exit $p\approx1$) pretend $\Delta P = 0$; otherwise integrate --- analytically at
$\delta = 0$ (\S4.6), numerically otherwise (Q19, \S4.13).
\end{itemize}

\spsub{4.10 \enspace Worked from the deck: ethane cracking in a PFR (Q17, slides~44, 85--88)}
\begin{qbox}
\textbf{Problem (slide~44).} Thermal cracking C$_2$H$_6 \to$ C$_2$H$_4 +$ H$_2$, first order,

\textbf{Step 1 --- kinetics:}
$k_{1100} = 0.072\exp\!\Big(\dfrac{82\,000}{1.987}\Big(\dfrac{1}{1000}-\dfrac{1}{1100}\Big)\Big)
= 3.067\ \mathrm{s^{-1}}$ --- the Arrhenius step moves $k$ by a factor of $43$ for $100$~K.\\[3pt]

(the slide~88 code writes it as \texttt{epsilon = yao*(1+1-1)}), so $\varepsilon = 1$ and
$V = \dfrac{v_0}{k}\Big[2\ln\dfrac{1}{1-X} - X\Big]$ (\S4.4).\\[3pt]
\textbf{Step 4 --- size:} at $X = 0.8$:
$V = \dfrac{2.318}{3.067}\big[2\ln 5 - 0.8\big] = 0.7558\times2.4183

\spsub{4.12 \enspace Worked from the deck: batch series $A\to B\to C$ (Q20, slides~77--79)}
\begin{qbox}
\textbf{Problem (slide~78).} The isothermal series reaction $A\to B\to C$ runs in a batch reactor with
$k_1 = 1\ \mathrm{h^{-1}}$, $k_2 = 2\ \mathrm{h^{-1}}$, and at $t = 0$: $C_A = 5$, $C_B = C_C = 0$ (the slide prints
$\mathrm{mol/m^3}$). Solve for the three concentrations and plot them versus time.
\end{qbox}
\begin{mthbox}
\textbf{The ODE system the deck codes} (MATLAB \texttt{ode45}, slides~78--79):
\[
\frac{dC_A}{dt} = -k_1 C_A,\qquad
\frac{dC_B}{dt} = k_1 C_A - k_2 C_B,\qquad
\frac{dC_C}{dt} = k_2 C_B .
\]
Exact solution: $C_A = C_{A0}e^{-k_1 t}$, $C_B = C_{A0}\dfrac{k_1}{k_2-k_1}\big(e^{-k_1 t}-e^{-k_2 t}\big)$,
$C_C = C_{A0}-C_A-C_B$. \textbf{The intermediate maximum:} set $dC_B/dt = 0$:
\[
\frac{k_1}{k_2-k_1}\big(-k_1 e^{-k_1 t}+k_2 e^{-k_2 t}\big) = 0
\ \Longrightarrow\ e^{(k_2-k_1)t} = \frac{k_2}{k_1}
\ \Longrightarrow\
t_{\max} = \frac{\ln(k_2/k_1)}{k_2-k_1} = \boxed{\;0.693\ \mathrm{h}\;},
\]
and back-substituting,
\[
C_{B,\max} = C_{A0}\left(\frac{k_1}{k_2}\right)^{k_2/(k_2-k_1)}
= 5\times(0.5)^{2} = \boxed{\;1.250\;}
\]
(B reaches only a quarter of the initial A --- with $k_2 > k_1$ the intermediate is destroyed about as fast as
the first step can make it; for a \emph{selective} series route one wants $k_2 \ll k_1$).
\end{mthbox}
\notetext{The deck's plotting block (slide~79): \texttt{plot(t,c(:,1),'+',t,c(:,2),'*',t,c(:,3))},
\texttt{legend('ca','cb','cc')}, \texttt{xlabel('time(hr)')}, \texttt{ylabel('concs, mol/m3')},
\texttt{title('Batch reactor')}. The printed $C_{A0} = 5\ \mathrm{mol/m^3}$ is dimensionally odd for a liquid
batch but is the scale the code and plot use; every \emph{shape} and both boxed results scale linearly or not at
all with it ($C_{B,\max}$ is linear in $C_{A0}$, $t_{\max}$ independent). The \texttt{ode45('batch',...)} call on
slide~79 is old MATLAB string syntax; modern versions want the handle \texttt{@batch}.}

\spsub{4.13 \enspace Worked from the deck: CSTR and PBR in either order (Q19, slides~74--77)}
\begin{qbox}
\textbf{Problem (slide~76).} The irreversible gas-phase reaction $A\to B$, \textbf{first order in the partial
pressure of A}, runs isothermally in a \textbf{fluidized} catalytic reactor holding 50~kg of catalyst. At present
$X = 0.50$ is realized with pure A entering at $20$~atm --- \emph{there is virtually no pressure drop in the CSTR}.
It is proposed to put a PBR of the same catalyst weight in series with the CSTR. For the PBR,
$\alpha = 0.018\ \mathrm{kg^{-1}}$ (particle size 0.2~mm, bed porosity 40~\%, viscosity as air at
$200\,^{\circ}$C). (a) PBR upstream or downstream of the CSTR for the highest exit conversion?
(b) The exit conversion of the last reactor? (c) The pressure at the exit of the packed bed? (d) How do the
answers change if the catalyst diameter is halved and the PBR diameter increased by 50~\%, assuming turbulent flow?
\end{qbox}
\begin{mthbox}
\textbf{Model (read the slide literally).} $A\to B$ has $\delta = 0$, so $\varepsilon = 0$ and
$p_A = P(1-X)$: the rate $-r_A' = k'p_A = k'P_0(1-X)p$ gives the coupled pair
\[
\frac{dX}{dW} = a\,(1-X)\,p\,,\qquad \frac{dp}{dW} = -\frac{\alpha}{2p}\,,\qquad
a \equiv \frac{k'P_0}{F_{A0}}\ \ [\mathrm{kg^{-1}}].
\]
\textbf{Calibration from the existing plant:} the fluidized reactor \emph{is} the CSTR (no drop, $p = 1$), so
$W = F_{A0}X/(k'P_0(1-X))$ at $X = 0.50$, $W = 50$~kg fixes
\[
a = \frac{X}{W(1-X)} = \frac{0.5}{50\times0.5} = 0.0200\ \mathrm{kg^{-1}}\quad(\text{exact}).
\]
\textbf{(c)} At $\delta = 0$ the exit pressure is model-independent:
$p_{\text{exit}} = (1-\alpha W)^{1/2} = (1-0.9)^{1/2} = 0.3162$, i.e.\ $P = 20\times0.3162
= \boxed{\;6.32\ \mathrm{atm}\;}$.\\[3pt]
\textbf{(a, b)} RK-4 integration of the pair ($h = 0.05$~kg):
\begin{itemize}
\item \textbf{CSTR then PBR} (PBR downstream): the CSTR delivers $X = 0.5$ at $p = 1$; the PBR continues to
$X = \mathbf{0.756}$ (its own exit $p = 0.3162$).
\item \textbf{PBR then CSTR}: the PBR alone reaches $X = 0.512$ at $p = 0.3162$; the trailing CSTR must then work
at 32~\% of full pressure and finishes at $X = \mathbf{0.629}$.
\end{itemize}
\textbf{Answer (a): the PBR goes downstream (CSTR first).} \textbf{(b): $X_{\text{exit}} = 0.756$.}\\[3pt]
\textbf{(d)} From the deck's own $\alpha = 2\beta_0/[A_c(1-\phi)\rho_c P_0]$ with turbulent Ergun
$\beta_0 \propto G^{2}/D_p$ and $G = \dot m/A_c$:
\[
\alpha \;\propto\; \frac{1}{A_c^{3}\,D_p}\,.
\]
$D_p \to D_p/2$ gives $\times 2$; tube diameter $\times1.5 \Rightarrow A_c\times2.25$ gives $\times 2.25^{-3}$:
\[
\alpha' = 0.018\times\frac{2}{2.25^{3}} = 0.00316\ \mathrm{kg^{-1}} .
\]
Re-integrating: CSTR then PBR $X = 0.808$ (from 0.756); PBR then CSTR $X = 0.800$ (from 0.629); the two
arrangements nearly converge and the packed-bed exit pressure rises to $p = 0.918$ ($18.4$~atm).
\end{mthbox}
\begin{daybox}
\textbf{The $\alpha$-scaling of part (d) is easy to half-miss.} The friction group $\beta_0$ itself sees only
$G^2/D_p$ ($\times 2$ and $\times 2.25^{-2}$ would give $\alpha' = 0.00711\ \mathrm{kg^{-1}}$), but converting
the pressure gradient to a drop \emph{per mass of catalyst} contributes the extra $1/A_c$ of the wider, shorter
bed --- the full $2.25^{-3}$ and $\alpha' = 0.00316\ \mathrm{kg^{-1}}$. And note (c) is purely geometric:
$p = \sqrt{1-\alpha W}$ carries no kinetics at all.
\end{daybox}

\toprule
Arrangement & \multicolumn{1}{c}{$a$ ($\mathrm{kg^{-1}}$)} & \multicolumn{1}{c}{$X_{\text{exit}}$}\\
\midrule
PBR then CSTR & 0.0200 & 0.629\\
CSTR then PBR & 0.0200 & 0.756\\
PBR then CSTR, $\alpha' = 0.00316$ & 0.0200 & 0.800\\

\notetext{Why CSTR-first wins: the CSTR converts at the \emph{inlet} pressure and its rate is first order in $p$;
the PBR then scavenges the remainder in plug flow with gradually falling $p$. In the reverse order the PBR burns
its volume near $X \approx 0$ (fast) but exits at $p = 0.32$, and the CSTR is stuck at that low pressure for its

\notetext{Source: \texttt{Ch5\_-\_krd.pdf}.}

\spsub{5.1 \enspace Differential method}
\begin{itemize}
\item Take logs of the power law to linearise it:
\end{itemize}
\begin{formulabox}
\begin{align*}
\ln\!\left(-\frac{dC_A}{dt}\right) &= \ln k + \alpha \ln C_A
 &&\text{slope } = \alpha,\ \text{intercept } = \ln k
\end{align*}
\end{formulabox}
\notetext{The derivatives come from the concentration--time data by finite differences:}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\toprule
initial point & $\dfrac{-3C_0 + 4C_1 - C_2}{2\Delta t}$\\
interior point & $\dfrac{C_{i-1} - C_{i+1}}{2\Delta t}$\\
final point & $\dfrac{C_{N-2} - 4C_{N-1} + 3C_N}{2\Delta t}$\\
\bottomrule
\end{tabularx}

\spsub{5.2 \enspace Integral method}
\begin{itemize}
\item \textbf{Guess the order, integrate, and test for a straight line.} The order that linearises the data is the order.
\end{itemize}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\toprule
order 0 & $C_A$ against $t$ is linear\\
order 1 & $\ln C_A$ against $t$ is linear\\
order 2 & $1/C_A$ against $t$ is linear\\
$A+B$, $M = C_{B0}/C_{A0} \ne 1$ & $\ln\dfrac{C_A}{C_B} = \ln M + C_{A0}(M-1)kt$\\
\bottomrule
\end{tabularx}

\spsub{5.3 \enspace Half-life method}
\begin{formulabox}
\begin{align*}
t_{1/2} &= \frac{2^{\,n-1}-1}{(n-1)\,k}\;C_{A0}^{\,1-n}
 &&\text{order } n \ne 1\\[5pt]
\ln t_{1/2} &= (1-n)\ln C_{A0} + \text{const}
 &&\text{slope } = 1-n \;\Longrightarrow\; n = 1-\text{slope}\\[4pt]
n=1:\quad t_{1/2} &= \frac{\ln 2}{k}
 &&\text{independent of } C_{A0}
\end{align*}
\end{formulabox}
\notetext{A half-life that does not change with initial concentration is the signature of first-order kinetics --- no regression needed.}

\spsub{5.4 \enspace Differential reactor and initial rates}
\begin{formulabox}
\begin{align*}
-r_A' = \frac{F_{A0}-F_A}{W} \approx \frac{F_P}{W}
 &&\text{low conversion, so composition is nearly uniform}
\end{align*}
\end{formulabox}

\spsub{5.5 \enspace Nonlinear regression and enzyme kinetics}
\begin{formulabox}
\begin{align*}
s^{2} &= \sum_i \left(C_{m,i} - C_{c,i}\right)^{2}
 &&\text{minimise; } m = \text{measured}, c = \text{calculated}\\[5pt]
-r_S &= \frac{k_3 C_S C_E}{C_S + C_M}
 &&\text{Michaelis--Menten}\\[5pt]
k_3 C_E t &= \left(C_{S0} - C_S\right) + C_M \ln\frac{C_{S0}}{C_S}
 &&\text{integrated form}
\end{align*}
\end{formulabox}

\spsub{5.6 \enspace Variable-volume batch, constant pressure}

\spsection{6 \enspace Multiple Reactions}
\notetext{Source: \texttt{Ch6\_krd.pdf}.}

\spsub{6.1 \enspace The four classes of multiple reaction}
\begin{itemize}
\item \textbf{Parallel (competing):} $A \xrightarrow{k_1} B$ and $A \xrightarrow{k_2} C$ consume the same feedstock at once; the reactor's job is to steer consumption toward one branch.
\item \textbf{Series (consecutive):} $A \xrightarrow{k_1} B \xrightarrow{k_2} C$ --- the product of the first reaction is the feed of the second, so the intermediate $B$ passes through a maximum.
\item \textbf{Complex (parallel $+$ series):} e.g.\ $A+B \xrightarrow{k_1} C+D$ together with $A+C \xrightarrow{k_2} E$; the deck also prints the \textbf{Denbigh network} $A \xrightarrow{1} R \xrightarrow{3} S$ with the side arrows $A \xrightarrow{2} T$ and $R \xrightarrow{4} U$ --- the standard bookkeeping test for net rates.
\item \textbf{Independent:} $A \to B+C$ and $D \to E+F$ run side by side without sharing species.
\end{itemize}
\notetext{The deck's own chemistry: parallel --- ethylene $+$ oxygen burning to $2\mathrm{CO_2}+2\mathrm{H_2O}$ ($k_1$) versus epoxidation to ethylene oxide ($k_2$); series --- ethylene oxide $+$ NH$_3$ to mono-, then di-, then tri-ethanolamine; complex --- ethanol to ethylene and to acetaldehyde, then ethylene $+$ acetaldehyde to butadiene $+$ water; independent --- cracking of crude, $\mathrm{C_{15}H_{32}} \to \mathrm{C_{12}H_{26}} + \mathrm{C_3H_6}$ and $\mathrm{C_8H_{18}} \to \mathrm{C_6H_{14}} + \mathrm{C_2H_4}$.}
With two reactants the deck's first lever is the \emph{contacting pattern} (slide~5):
\begin{itemize}
\item $C_A,\,C_B$ both high --- charge $A$ and $B$ all at once (batch);
\item $C_A,\,C_B$ both low --- add both slowly to a well-mixed vessel (semibatch / fed CSTR);
\item $C_B$ high, $C_A$ low --- start with $B$ and feed $A$ slowly.
\end{itemize}
\notetext{The economic incentive (slide~8): with $D$ desired and $U$ undesired, maximise $D$ and minimise $U$; the deck's cost sketch shows the reaction cost falling and the separation cost rising as the system is pushed harder, so the \emph{total} cost has a minimum in between --- selectivity, not conversion alone, sets the optimum.}

\spsub{6.2 \enspace Selectivity and yield --- the four definitions}
\begin{formulabox}
\begin{align*}
S_{D/U} &= \frac{r_D}{r_U}
 &&\text{instantaneous (rate) selectivity}\\[3pt]
\tilde{S}_{D/U} &= \frac{F_D}{F_U},\ \ \frac{N_D}{N_U}\ \text{(batch)}
 &&\text{overall selectivity}\\[4pt]
Y_D &= \frac{r_D}{-r_A}
 &&\text{instantaneous yield}\\[3pt]
\tilde{Y}_D &= \frac{F_D}{F_{A0}-F_A},\ \ \frac{N_D}{N_{A0}-N_A}\ \text{(batch)}
 &&\text{overall yield}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{In a CSTR the two selectivities coincide} (slide~11): $F_D = r_D V$ and $F_U = r_U V$, so $\tilde S_{D/U} = r_D V / r_U V = S_{D/U}$.
\item \textbf{Definitions differ between authors} --- slide~14 says to check which one a question intends. The \emph{overall} quantities set the profit of the reaction--separation system; the \emph{instantaneous} ones are the design tool, because they say which reactor and contacting scheme to choose.
\end{itemize}

\spsub{6.3 \enspace The rate selectivity parameter, one reactant}
\notetext{For $A \xrightarrow{k_D} D$ (desired) and $A \xrightarrow{k_U} U$ (undesired) with $-r_A = k_D C_A^{\alpha_1} + k_U C_A^{\alpha_2}$:}
\begin{formulabox}
\begin{align*}
S_{D/U} &= \frac{r_D}{r_U} = \frac{k_D}{k_U}\,C_A^{\,\alpha_1-\alpha_2}
 &&\text{maximise this}\\[4pt]
\frac{k_D}{k_U} &= \frac{A_D}{A_U}\,e^{-(E_D-E_U)/RT}
 &&\text{the temperature lever}
\end{align*}
\end{formulabox}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\toprule
\textbf{Case 1} $\alpha_1 > \alpha_2$ & keep $C_A$ \textbf{high}: batch or plug flow, no inerts and high pressure (gas), minimum diluent (liquid)\\\\
\textbf{Case 2} $\alpha_2 > \alpha_1$ & keep $C_A$ \textbf{low}: CSTR or recycle reactor (the product stream acts as diluent), inerts and low pressure (gas), maximum diluent (liquid)\\\\
\textbf{Case 3} $E_D > E_U$ & run \textbf{hot}: $k_D$ grows faster with $T$ than $k_U$\\\\
\textbf{Case 4} $E_U > E_D$ & run \textbf{cool} --- but not so cool that the desired reaction stops proceeding\\\\
\bottomrule
\end{tabularx}
\notetext{The deck's rules of thumb (slide~4) are the same table in words: a \emph{high} reactant concentration favours the reaction of \emph{higher} order, a \emph{low} concentration the reaction of \emph{lower} order, and for reactions of the \emph{same} order concentration has no effect on the distribution.}
\begin{itemize}
\item \textbf{Worked selectivity argument} (slides~18--19). Three parallel paths from $A$:
\end{itemize}
\begin{align*}
r_D &= 0.0002\,e^{36\,000\left(\frac{1}{300}-\frac{1}{T}\right)} C_A &
r_U &= 0.0018\,e^{25\,000\left(\frac{1}{300}-\frac{1}{T}\right)} C_A^{1.5} &
r_Q &= 0.00452\,e^{5\,000\left(\frac{1}{300}-\frac{1}{T}\right)} C_A^{0.5}
\end{align*}
\begin{mthbox}
\begin{align*}
S_{D/Q} &= \frac{r_D}{r_Q} \approx \text{very large at high } T
 &&(E_D = 36\,000 \gg E_Q = 5\,000)\\[3pt]
S_{D/U} &= \frac{r_D}{r_U}
 = \frac{0.11\,e^{11\,000\left(\frac{1}{300}-\frac{1}{T}\right)}}{C_A^{\,0.5}}
 &&\left(\tfrac{0.0002}{0.0018} = 0.11,\ \ 36\,000-25\,000 = 11\,000,\ \ 1-1.5 = -\tfrac12\right)
\end{align*}
\end{mthbox}
\notetext{Two levers survive the algebra: \textbf{high temperature} (to bury $Q$) and \textbf{low concentration of $A$} (to bury $U$) --- the latter by inerts, low pressure, or a CSTR / recycle reactor. One printed sentence runs against the algebra: slide~18's ``$S_{DU}$ is minimized at low concentration of $A$'' states the reverse of the $C_A^{-0.5}$ dependence of the slide's own equation~(5); slide~19's operating conclusions agree with the algebra, and the sentence is reported as printed.}

\spsub{6.4 \enspace Two reactants: signs, schemes and the four cases}
\notetext{For $A+B \xrightarrow{k_1} D$ and $A+B \xrightarrow{k_2} U$ with $r_D = k_1 C_A^{\alpha_1}C_B^{\beta_1}$, $r_U = k_2 C_A^{\alpha_2}C_B^{\beta_2}$:}
\begin{formulabox}
\begin{align*}
S_{D/U} &= \frac{k_1}{k_2}\,C_A^{\,\alpha_1-\alpha_2}\,C_B^{\,\beta_1-\beta_2}
\end{align*}
\end{formulabox}
\notetext{The deck's menu of contacting schemes (slide~21): (a)~CSTR; (b)~tubular reactor; (c)~batch; (d)~semibatch, B fed slowly into A; (e)~semibatch, A fed slowly into B; (f)~tubular reactor with side streams of B; (g)~tubular reactor with side streams of A; (h)~series of small CSTRs; (i)~tubular reactor with recycle.}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\toprule
\textbf{Case I} $\alpha_1>\alpha_2,\ \beta_1>\beta_2$ & keep $C_A,C_B$ \textbf{both high} --- (b) tubular, (c) batch, high pressure, reduce inert\\\\
\textbf{Case II} $\alpha_1>\alpha_2,\ \beta_1<\beta_2$ & $C_A$ high, $C_B$ low --- (d) semibatch B into A, (f) side streams of B, (h) small CSTRs with A fed only to the first\\\\
\textbf{Case III} $\alpha_1<\alpha_2,\ \beta_1<\beta_2$ & keep $C_A,C_B$ \textbf{both low} --- (a) CSTR, (i) large recycle ratio, feed diluted with inert, low pressure\\\\
\textbf{Case IV} $\alpha_1<\alpha_2,\ \beta_1>\beta_2$ & $C_A$ low, $C_B$ high --- (e) semibatch A into B, (g) side streams of A, (h) small CSTRs with fresh B to each\\\\
\bottomrule
\end{tabularx}

\end{warnbox}

\spsub{6.5 \enspace Series reactions --- maximising the intermediate}
\begin{formulabox}
\begin{align*}
A \;\xrightarrow{\;k_1\;}\; B \;\xrightarrow{\;k_2\;}\; C
\qquad\Longrightarrow\qquad
C_B &= C_{A0}\,\frac{k_1}{k_2-k_1}\left(e^{-k_1t}-e^{-k_2t}\right)\\[3pt]
t_{\text{opt}} &= \frac{\ln(k_2/k_1)}{k_2-k_1}
 &&\text{maximises } C_B
\end{align*}
\end{formulabox}
\begin{itemize}
\item In \textbf{parallel} networks the product is maximised by adjusting the reaction \emph{conditions} and choosing the proper reactor; in \textbf{series} networks the lever is \emph{time} --- the space time of a flow reactor or the real time of a batch. The $C_B(t)$ form above is the one behind the Ch4 series numerical reproduced in \S4.12 (Q20).
\item The deck's triangular diagram (slide~27): $k_1/k_2 \gg 1$ gives a large quantity of $B$; $k_1/k_2 \sim 1$ a moderate one; $k_1/k_2 \ll 1$ (first reaction slow, second fast) very little. In every case a \emph{long} reaction time in batch or a \emph{long} tubular reactor converts $B$ onward to $C$ --- which is why the optimum is interior.
\item Verified numerically in \S4.12: $t_{\max} = 0.693$~h gives $C_{B,\max} = 1.250$.
\end{itemize}

\spsub{6.6 \enspace Product distribution --- the quantitative treatment}
\begin{formulabox}
\begin{align*}
\varphi &= \frac{dC_R}{-dC_A}
 &&\text{instantaneous fractional yield of } R\\[4pt]
\Phi &= \frac{C_{Rf}}{C_{A0}-C_{Af}}
 &&\text{overall yield} = \bar\varphi \text{ in the reactor}\\[4pt]
\text{PFR:}\quad \Phi_p &= \frac{1}{\Delta C_A}\int_{C_{A0}}^{C_{Af}} \varphi\,dC_A
 &&\text{area under } \varphi(C_A)\\[4pt]
\text{MFR:}\quad \Phi_m &= \varphi \text{ evaluated at } C_{Af}
 &&\text{one rectangle}\\[4pt]
\text{$N$ staged MFRs:}\quad
\Phi_{N} &= \frac{\sum_i \varphi_i\left(C_{A,i-1}-C_{A,i}\right)}{C_{A0}-C_{AN}}
 &&\text{steps between the two}\\[4pt]
C_{Rf} &= \Phi\left(C_{A0}-C_{Af}\right)
 &&\text{product out of any reactor}
\end{align*}
\end{formulabox}
\notetext{Slide~31 draws the three constructions on one $\varphi$ curve: plug flow shades the area under the curve from $C_{Af}$ to $C_{A0}$, mixed flow draws the single rectangle of height $\varphi(C_{Af})$ over $(-\Delta C_A)$, and staged reactors climb the curve in steps --- the rectangle construction converging to the area as $N \to \infty$.}

\spsub{6.7 \enspace Net rates and the bookkeeping for $q$ reactions}
\begin{formulabox}
\begin{align*}
r_j &= \sum_{i=1}^{q} r_{ij}
 &&\text{net rate of species } j = \text{sum over reactions } i
\end{align*}
\end{formulabox}
\begin{itemize}
\item The deck's example set (slide~34): reaction~1, $A+B \to 3C+D$; reaction~2, $A+2C \to 3E$; reaction~3, $2B+3E \to 4F$; reaction~$q$, $A+\tfrac12 C \to G$. Then $r_A = r_{1A}+r_{2A}+r_{3A}+\cdots+r_{qA}$ while $r_B$ picks up only reactions 1 and~3 --- species absent from a reaction contribute zero to its sum.
\end{itemize}

\spsub{6.8 \enspace Worked from the deck}
\begin{itemize}
\item \textbf{Product distribution} (slide~32). $dC_R/dt = 1.0\,C_A^{1.5}C_B^{0.3}$, $dC_S/dt = 1.0\,C_A^{0.5}C_B^{1.8}$; equal volumetric feeds each at $20$~mol/L, $90\,\%$ conversion of A. The same chemistry through three contacting patterns:
\[
\text{(a) plug flow } C_R = 2.859 \qquad
\text{(b) mixed flow } C_R = 4.50 \qquad
\text{(c) plug flow, B in side streams } C_R = 7.295
\]
\notetext{With equal feeds $C_A = C_B$ everywhere, so $r_R/r_S = C^{-0.5}$: \emph{low} concentration favours R. Mixed flow then runs entirely at the lowest concentration, plug flow must pass through the high-concentration region making S, and side entrances of B depress $C_B$ relative to $C_A$ further still --- hence $2.86 < 4.50 < 7.30$. A factor of $2.6$ in product from contacting alone, with no change to chemistry, temperature or throughput.}

\spsection{7 \enspace Non-Isothermal Reactor Design}
\notetext{Source: \texttt{Ch7-\_krd.pdf}.}

\spsub{7.1 \enspace Performance equations and the energy balance}
\begin{itemize}
\item The isothermal algorithm of section~4 survives unchanged; temperature only adds one unknown, coupled in through Arrhenius. Slide~5 writes the set for a first-order $A\to B$ PFR:
\end{itemize}
\begin{formulabox}
\begin{align*}
\frac{dX}{dV} &= \frac{-r_A}{F_{A0}}, &
-r_A &= k_1 \exp\!\left[\frac{E}{R}\left(\frac{1}{T_1}-\frac{1}{T}\right)\right] C_{A0}(1-X)
\end{align*}
\end{formulabox}
\begin{itemize}
\item The open-system energy balance (slides~6--8) reads \emph{accumulation $=$ heat added $-$ work done $+$ energy in by flow $-$ energy out by flow}. With $E_i \approx U_i$ (potential and kinetic energies dropped) and flow work absorbed into the enthalpy, steady state gives
\end{itemize}
\begin{formulabox}
\begin{align*}
\dot{Q} - \dot{W}_s + \sum F_{i0}H_{i0} - \sum F_i H_i &= 0\\[4pt]
H_i(T) &= H_i^{\circ}(T_R) + C_{p,i}\left(T - T_R\right)
 &&\text{no phase change, constant } C_p
\end{align*}
\end{formulabox}
\begin{itemize}
\item Substituting the stoichiometry $F_i = F_{A0}(\Theta_i + \nu_i X/a)$ collapses everything to the deck's master equation (slides~9--11):
\end{itemize}
\begin{formulabox}
\begin{align*}
\dot{Q} - \dot{W}_s - F_{A0}X\left[\Delta H_{Rx}^{\circ}(T_R) + \Delta\hat{C}_p\left(T-T_R\right)\right]
 - F_{A0}\sum_i \Theta_i \tilde{C}_{p,i}\left(T - T_{i0}\right) &= 0\\[4pt]
\Delta\hat{C}_p &= \tfrac{d}{a}C_{pD} + \tfrac{c}{a}C_{pC} - \tfrac{b}{a}C_{pB} - C_{pA}
\end{align*}
\end{formulabox}

\spsub{7.2 \enspace Adiabatic operation}
\begin{formulabox}
\begin{align*}
X &= \frac{\sum \Theta_i \tilde{C}_{p,i}\left(T-T_0\right)}
        {-\left[\Delta H_{Rx}^{\circ}(T_R) + \Delta\hat{C}_p\left(T-T_R\right)\right]}
 &&\text{the energy-balance line } X_{EB}\\[5pt]
T &= \frac{X\left[-\Delta H_{Rx}^{\circ}(T_R)\right] + \sum\Theta_i C_{p,i}T_{i0} + X\Delta C_p T_R}
        {\sum\Theta_i C_{p,i} + X\Delta C_p}
 &&\text{solved for } T
\end{align*}
\end{formulabox}
\begin{itemize}
\item The same straight line $X_{EB}$ on the $T$--$X$ plane holds for \textbf{batch, CSTR, PFR and PBR} --- the deck brackets all four beside the plot. For an exothermic reaction $X_{EB}$ rises with $T$; the reactor can only operate where this line meets what the kinetics and equilibrium allow.
\end{itemize}

\spsub{7.3 \enspace The equilibrium ceiling, from thermodynamics}
\notetext{Slides~35--45 develop the ceiling rather than asserting it:}
\begin{formulabox}
\begin{align*}
dG &= Vdp - SdT + \textstyle\sum_j \mu_j\,dn_j, &
\mu_j &= \left(\frac{\partial G}{\partial n_j}\right)_{T,P,n_i}\\[4pt]
\Delta G &= \Delta G^{\circ} + RT\ln\!\left(\textstyle\prod_j a_j^{\nu_j}\right), &
K_a &= \left(\textstyle\prod_j a_j^{\nu_j}\right)_{eq} = \exp\!\left(\frac{-\Delta G^{\circ}}{RT}\right)\\[4pt]
K_p &= \left(\textstyle\prod_j P_j^{\nu_j}\right),\quad
K_f = \left(\textstyle\prod_j f_j^{\nu_j}\right),\quad
K_C = \left(\textstyle\prod_j C_j^{\nu_j}\right)
 &&\text{activity specialised}\\[4pt]
\frac{d\ln K_a(T)}{dT} &= \frac{\Delta H}{RT^{2}}
 &&\text{van 't Hoff}\\[4pt]
K_p &= \left(\textstyle\prod_j \left[P\,\dfrac{N_{j0}+\nu_j\alpha}{N_{T0}+\alpha\sum_j\nu_j}\right]^{\nu_j}\right) = F(\alpha)
 &&\text{extent form}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{Reversible exothermic:} $X_e$ falls with $T$ (the equilibrium curve descends); \textbf{reversible endothermic:} $X_e$ rises with $T$. The adiabatic energy-balance line crosses the equilibrium curve; that intersection is the \emph{adiabatic equilibrium conversion}, and no catalyst moves it (slide~46).
\item \textbf{Worked intersection} (slides~47--48): $A \rightleftharpoons B$, pure A at $300$~K, $H_{A0} = -40\,000$, $H_{B0} = -60\,000$~cal/mol, $C_{pA} = C_{pB} = 50$~cal/mol\,K, $K_C(298) = 10^{5}$:
\end{itemize}
\begin{mthbox}
\begin{align*}
X_{EB} &= \frac{C_p(T-T_0)}{-\Delta H_{Rx}} = 2.5\times10^{-3}(T-300)
 &&\left(\tfrac{50}{20\,000}\right)\\[3pt]
K_C(T) &= 100\,000\,\exp\!\left[-33.78\,\frac{T-298}{T}\right]
 &&\left(\tfrac{|\Delta H|}{R\cdot 298} = \tfrac{10\,065}{298} = 33.78\right)\\[3pt]
X_e &= \frac{K_C(T)}{1+K_C(T)}
 &&\text{intersection: deck graphic } 0.42 \text{ at } 465\ \mathrm{K}
\end{align*}
\end{mthbox}
\notetext{Solving the two printed equations exactly gives $X = 0.401$ at $460.4$~K --- the deck's $0.42/465$~K is a graphical read-off of the same pair, reproduced by \texttt{krd-answers-check.py}. The deck's $K_C$ table (slide~47: $661.6$ at $350$~K down to $0.12$ at $500$~K) is the same formula evaluated.}

\spsub{7.4 \enspace Heat exchange in a tubular reactor}
\begin{formulabox}
\begin{align*}
\frac{dT}{dV} &= \frac{Ua\left(T_a - T\right) + (-r_A)(-\Delta H_{Rx})}{\sum F_i C_{p,i}}
 &&\text{no radial gradients}\\[5pt]
\text{co-current coolant:}\quad \frac{dT_a}{dV} &= \frac{Ua\left(T - T_a\right)}{\dot{m}_c C_{p,c}}
\end{align*}
\end{formulabox}
\begin{itemize}
\item If the coolant flow is large compared with the heat released, $T_a$ is effectively \textbf{constant} along the reactor (slide~26); heat added for endothermic, removed for exothermic.
\item \textbf{Acetic anhydride} (slides~27--30): acetone cracking $\mathrm{CH_3COCH_3} \to \mathrm{CH_2CO} + \mathrm{CH_4}$, first order with $\ln k = 34.34 - 34\,222/T$, $7850$~kg/h through a bank of $1000$ 1-inch schedule-40 tubes, inlet $1035$~K, $162$~kPa. Three cases: adiabatic --- the endothermic reaction cools itself and \emph{dies after $\approx 3.5$~m$^3$}; constant $T_a = 1150$~K with $U = 110$~J/m$^2$s\,K --- $T$ dips then recovers and $X$ climbs; variable heating medium --- $T_a$ slides down with $T$.
\item \textbf{Ethane cracking at equilibrium} (slide~31): ethane $+$ steam, $2.99$~atm, $880\,^{\circ}$C, $95$~m tubes, $60\,\%$ conversion --- the deck's specification sheet for the equilibrium-limited endothermic case.
\end{itemize}

\spsub{7.5 \enspace The CSTR with heat exchange --- two solution routes}
\notetext{With $\kappa = UA/(F_{A0}C_{p0})$ and $T_c = (\kappa T_a + T_0)/(1+\kappa)$:}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\toprule
\textbf{Specify $X$} & $T = T_c + \dfrac{(-\Delta H_{Rx}^{\circ})(X)}{C_{p0}(1+\kappa)}$, then $k = Ae^{-E/RT}$, $-r_A = kC_{A0}(1-X)$, $V = F_{A0}X/(-r_A)$\\\\
\textbf{Specify $T$} & $X = \dfrac{C_{p0}(1+\kappa)(T-T_c)}{-\Delta H_{Rx}^{\circ}}$, then the same rate-law chain to $V$\\\\
\textbf{Specify $V$} & generate $T$ vs $X_{EB}$ from the energy balance and $T$ vs $X_{MB}$ from $X_{MB} = \tau k/(1+\tau k)$; the intersection is the operating point\\\\
\bottomrule
\end{tabularx}
\begin{formulabox}
\begin{align*}
G(T) &= X(-\Delta H_{Rx}) = \frac{\tau A e^{-E/RT}}{1 + \tau A e^{-E/RT}}\,(-\Delta H_{Rx})
 &&\text{heat generated, first order}\\[4pt]
R(T) &= C_{p0}\left(1+\kappa\right)\left(T - T_c\right)
 &&\text{heat removed}
\end{align*}
\end{formulabox}
\begin{itemize}
\item Raising $T_0$ slides the straight $R(T)$ line to the right; raising $\kappa$ steepens it and moves $T_c$ from $T_0$ ($\kappa=0$) toward $T_a$ ($\kappa\to\infty$) (slides~77--78). Raising $\tau$ or $E$ sharpens the S-shape of $G(T)$ (slide~81) --- which is what creates multiple intersections.
\end{itemize}

\spsub{7.6 \enspace Multiple steady states, ignition--extinction, runaway}
\begin{itemize}
\item Intersections of $G(T)$ and $R(T)$ are the steady states; sweeping $T_0$ traces the \textbf{ignition--extinction curve}: upper and lower branches are attainable, the middle branch is not (slides~82--84).
\item \textbf{Stability} (slides~85--89): a state is stable where a temperature bump removes more heat than it generates. The deck's criterion: with
\[
\Delta T_{bc} = T' - T_c = \frac{RT_c^{\,2}}{E},\qquad
T' = \frac{E}{2R}\left[1 - \sqrt{1 - 4\tfrac{RT_c}{E}}\,\right],\qquad
T_c = \frac{\kappa T_a + T_0}{1+\kappa},
\]
exceeding $\Delta T_{bc}$ tips the reactor to the upper state. For industrial reactions $E/RT \approx 16$--$24$, so $\Delta T_{bc} \approx 15$--$30\,^{\circ}$C.
\item With $S' = dG/dT$ at the steady state, \textbf{runaway} when $C_{p0}(1+\kappa) < S'$ (slides~88, 90--91): decreasing $UA$ flattens $R(T)$ until tangency disappears and the CSTR runs away to the upper state. The deck demonstrates it on $2\mathrm{Na_2S_2O_3} + 4\mathrm{H_2O_2} \to \mathrm{Na_2S_3O_6} + \mathrm{Na_2SO_4} + 4\mathrm{H_2O}$ (slides~85--86).
\item \textbf{Zero vs first order} (slides~92--93): with $\tau = 0.1$~min, $E = 10\,000$~cal/mol, $\Delta H_{Rx} = -10\,000$~cal/mol and inerts $2{:}1$, the deck's table gives $dE/dT = 80$; the first-order reaction is globally stable for all $T_c$, while the zero-order one runs away above an entering temperature of $460$~K and becomes stable again only past $685$~K.
\end{itemize}

\spsub{7.7 \enspace Interstage cooling, interstage heating, optimum feed temperature}
\begin{itemize}
\item \textbf{Exothermic reversible:} several adiabatic stages with \emph{coolers} between them; each stage starts cooler, further below its equilibrium curve, and the zigzag path on the $X$--$T$ diagram climbs above what one adiabatic bed can reach (slides~49, 52).
\item \textbf{Endothermic reversible:} mirror image --- \emph{heaters} between stages (slide~50).
\item \textbf{Optimum feed temperature} (slides~53--54): at low $T_0$ the rate is slow and conversion is low; at high $T_0$ the equilibrium conversion collapses and conversion is again low; between them $X(T_0)$ passes through a \emph{maximum} --- the optimum feed temperature for a fixed size or catalyst weight.
\end{itemize}

\spsection{8 \enspace Catalysis and Catalytic Reactors}
\notetext{Source: \texttt{Catalysis\_Krd.pdf}.}

\spsub{8.1 \enspace What a catalyst does --- and what it cannot do}
\begin{itemize}
\item \textbf{Catalyst} (slide~5): a substance that increases the \emph{rate} at which a chemical system approaches equilibrium, \emph{without being substantially consumed}. Selecting one is \emph{not well understood} --- in practice, extensive trial and error.
\item \textbf{Homogeneous} --- catalyst in the same phase as the reactants: $2\,\mathrm{SO_2(g)}+\mathrm{O_2(g)} \xrightarrow{\mathrm{NO(g)}} 2\,\mathrm{SO_3(g)}$; sucrose $+\,\mathrm{H_2O} \xrightarrow{\mathrm{H^+(aq)}}$ glucose $+$ fructose. \textbf{Heterogeneous} --- different phase: $4\,\mathrm{NH_3}+5\,\mathrm{O_2} \xrightarrow{\mathrm{Pt},\,800\,\degreec} 4\,\mathrm{NO}+6\,\mathrm{H_2O}$.
\end{itemize}
\begin{formulabox}
\textbf{The catalyst changes the PACE, not the thermodynamics} (slide~15): $\Delta G$ and $K_{eq}$ of the reaction are untouched. The deck's example, $\mathrm{CH_4(g)}+\mathrm{CO_2(g)} = 2\,\mathrm{CO(g)}+2\,\mathrm{H_2(g)}$:
\begin{align*}
\Delta G_{100\,\degreec} &= +151\ \mathrm{kJ/mol} &&\text{no driving force: no reaction, with or without catalyst}\\
\Delta G_{700\,\degreec} &= -16\ \mathrm{kJ/mol} &&\text{driving force present, but the mixture still does not react on its own}
\end{align*}
At $700\,\degreec$ with $\mathrm{Pt/ZrO_2}$ or $\mathrm{Ni/Al_2O_3}$ in the reactor, equilibrium conversion ($\sim$100\%) is reached at that same temperature.
\end{formulabox}

\spsub{8.2 \enspace Solid catalysts: active phase, promoters, supports}
\begin{itemize}
\item A solid catalyst is built from three roles (slide~16): the \textbf{active phase} where reaction occurs (mostly metal/metal oxide); a \textbf{promoter} --- textual (e.g.\ Al for NH$_3$ production), electric/structural, or poison-resistant; and a \textbf{support/carrier} giving mechanical strength and surface area (98\% of the surface is inside the porous structure; the support may or may not be catalytically active). \textbf{Gauze/foil}: external active sites alone are enough for very fast rates.
\item \textbf{Support materials} (slide~17): alumina --- inexpensive, $1$--$700$~m$^2$/g, acidic/neutral; silica --- $100$--$800$~m$^2$/g, acidic; \textbf{zeolite} --- mixture of alumina and silica, often exchanged metal ion present, \emph{shape selective}, acidic; others: active carbon (up to $1000$~m$^2$/g), titania ($10$--$50$), zirconia ($10$--$100$), magnesia ($10$), lanthana ($10$).
\item \textbf{Specificity} (slides~21, 28): the choice of catalyst selects the product. Ethanol over copper dehydrogenates, $\mathrm{C_2H_5OH} \to \mathrm{CH_3CHO}+\mathrm{H_2}$; over alumina it dehydrates, $\mathrm{C_2H_5OH} \to \mathrm{C_2H_4}+\mathrm{H_2O}$. The distance between active sites and their match to bond lengths decides which bonds are weakened.
\end{itemize}

\spsub{8.3 \enspace The seven steps; the three intrinsic stages}
\begin{daybox}
\textbf{External diffusion $\to$ internal diffusion $\to$ adsorption of A $\to$ surface reaction $\to$ desorption of B $\to$ internal diffusion out $\to$ external diffusion out.}
\vspace{2pt}
Assume one step limits, write its rate, and combine with the \textbf{site balance}
\[
C_t = C_v + \sum_i C_{i\cdot S}
\]
Then test the assumption: if the assumed rate-limiting step does not reproduce the data, move to the next one. The deck works through each assumption in turn.
\end{daybox}
\begin{itemize}
\item Inside the film and the pore the three intrinsic stages are \textbf{adsorption $\to$ reaction $\to$ desorption} (slides~24--25): adsorption lands the species on an active site and weakens its bonds; the surface holds adsorbed gases in the right orientation so favourable collisions become likely; desorption releases the products after an electron rearrangement.
\end{itemize}

\toprule
 & \textbf{Physisorption} & \textbf{Chemisorption}\\\\
\midrule
force & van der Waals & chemical bond\\\\
adsorbed layers & multilayer & only one layer\\\\
heat of adsorption & low ($10$--$40$~kJ/mol) & high ($>40$~kJ/mol)\\\\
selectivity & low & high\\\\
temperature at which it occurs & low & high\\\\
\bottomrule
\end{tabularx}
\notetext{Slide~27. Chemisorption is the catalytically useful one: a real bond to the surface is what activates the molecule.}

\spsub{8.4 \enspace Mechanisms: Langmuir--Hinshelwood and Eley--Rideal}
\begin{itemize}
\item \textbf{Langmuir--Hinshelwood} (slides~29--30): adsorption from the gas phase, reactions \emph{between adsorbed molecules}, dissociation at the surface, desorption. For $A_2 + 2B = 2AB$ the deck writes
\[
A_2 + * = A_2^*,\quad A_2^* + * = 2A^*,\quad B + * = B^*,\quad A^* + B^* = AB^* + *,\quad AB^* = AB + *
\]
\item \textbf{Eley--Rideal} (slides~31--32): the same adsorption steps, but the last step is the direct reaction between the adsorbed molecule $A^*$ and a \emph{gas-phase} molecule $B$: $A^* + B = AB + *$. That fourth step cannot occur in an L--H mechanism.
\item \textbf{Telling them apart} (slide~33), rate versus coverage $\theta_{A,S}$: for E--R the rate \emph{increases} with coverage until the surface is completely covered by $A^*$; for L--H the rate goes through a \emph{maximum} and ends at zero when $A^*$ blocks every site --- because the step $B + * = B^*$ requires a free site. Whether a real reaction follows one or the other cannot be decided without experimental or computational studies.
\end{itemize}
\begin{daybox}
\textbf{Intrinsic-kinetics algorithm} (slide~34): write a rate equation for every step; assume one step is the RDS --- the intrinsically slowest; set $r_o = r_{\text{RDS}}$; take all other steps at equilibrium ($r \approx 0$); express every surface-adsorbed species concentration in terms of bulk species; write the site balance; substitute the adsorbed-species equations and $C_v$ into $r_{\text{RDS}}$.
\end{daybox}
\begin{itemize}
\item \textbf{Ammonia synthesis mechanism as printed} (slide~36): $\mathrm{H_2}+2S \to 2\,\mathrm{H\!\cdot\!S}$ (rapid); $\mathrm{N_2}+S \rightleftharpoons \mathrm{N_2\!\cdot\!S}$ and $\mathrm{N_2\!\cdot\!S}+S \to 2\,\mathrm{N\!\cdot\!S}$ (\textbf{rate-limiting}); then $\mathrm{N\!\cdot\!S}+\mathrm{H\!\cdot\!S} \rightleftharpoons \mathrm{HN\!\cdot\!S}+S$, $\mathrm{NH\!\cdot\!S}+\mathrm{H\!\cdot\!S} \rightleftharpoons \mathrm{H_2N\!\cdot\!S}+S$, $\mathrm{H_2N\!\cdot\!S}+\mathrm{H\!\cdot\!S} \rightleftharpoons \mathrm{NH_3\!\cdot\!S}+S$, $\mathrm{NH_3\!\cdot\!S} \rightleftharpoons \mathrm{NH_3}+S$ (rapid).
\item \textbf{Cyclohexane dehydrogenation} (slide~35): $\mathrm{C_6H_{12}} \xrightarrow[\mathrm{Al_2O_3}]{\mathrm{Ni}} \mathrm{C_6H_6} + 3\,\mathrm{H_2}$. The measured rate rises with concentration and saturates --- the signature of an adsorption denominator --- and the task is to (1)~propose the mechanism and (2)~find the RDS consistent with the data.
\end{itemize}

\spsub{8.5 \enspace Adsorption isotherms and rate forms}
\begin{formulabox}
\begin{align*}
C_{A\cdot S} &= \frac{K_A P_A\, C_t}{1 + K_A P_A + K_B P_B}
 &&\text{Langmuir, competitive sites}\\[5pt]
-r_T' &= \frac{k\,P_T P_{\mathrm{H_2}}}{1 + K_B P_B + K_T P_T}
 &&\text{Eley--Rideal type}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{Langmuir--Hinshelwood} assumes both reactants are adsorbed before reacting; \textbf{Eley--Rideal} assumes one reacts directly from the gas phase. The denominator is the site balance in both cases --- that is what makes the rate saturate at high pressure.
\end{itemize}

\spsub{8.6 \enspace Pore structure and the effective diffusivity}
\begin{itemize}
\item \textbf{Pore classes} (slide~38): mesopores $<2$~nm; micropores $2$ to $50$~nm; macro pores $>50$~nm; the active component coats the pore walls. Pores are of random size except in zeolites.
\end{itemize}
\notetext{The deck's micro/meso labels are inverted relative to IUPAC (IUPAC: micro $<2$~nm, meso $2$--$50$~nm). Reported as printed, not silently corrected --- the sizes, not the names, are what the calculations use.}
\begin{formulabox}
\begin{align*}
D_e &= \frac{D_{AB}\,\phi_p\,\sigma_c}{\tilde\tau}
 &&\text{effective diffusivity}\\[4pt]
\tilde\tau &= \frac{\text{actual distance travelled}}{\text{shortest distance}}
 &&\text{tortuosity};\quad
\beta = \frac{\text{area } A_2}{\text{area } A_1},\quad \sigma_c = f(\beta)\\[4pt]
\text{90\textdegree\ bent pore:}\quad
\tilde\tau &= \frac{2L}{\sqrt{2L^2}} = \sqrt{2} = 1.414
 &&\text{worked on slide~40}
\end{align*}
\end{formulabox}

\spsub{8.7 \enspace Internal diffusion: Thiele modulus, effectiveness factor, geometries}
\begin{formulabox}
\begin{align*}
\frac{d^2\psi}{d\lambda^2} + \frac{2}{\lambda}\frac{d\psi}{d\lambda} - \phi_1^{2}\,\psi &= 0
 &&\psi = \frac{C_A}{C_{As}},\ \lambda = \frac{r}{R},\quad
 \phi_1^{2} = \frac{k_1 R^{2} C_{As}^{\,n-1}}{D_e}\quad\text{(slide~42)}\\[5pt]
\phi_s &= R\sqrt{\frac{k_1}{D_e}}
 &&\text{Thiele modulus, first order, sphere}\\[5pt]
\eta &= \frac{3}{\phi^{2}}\left(\phi\coth\phi - 1\right)
 &&\text{effectiveness factor, sphere}\\[4pt]
\eta &= \frac{\tanh\phi}{\phi}
 &&\text{slab geometry}\\[5pt]
\frac{C_A}{C_{As}} &= \frac{R}{r}\cdot\frac{\sinh(\phi\, r/R)}{\sinh\phi}
 &&\text{concentration profile inside the pellet}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{$\eta \to 1$} for small $\phi$ (reaction-limited: the whole pellet is used); \textbf{$\eta \to 3/\phi$} for large $\phi$ (diffusion-limited: only a shell near the surface reacts).
\item \textbf{Other geometries} (slide~44), with $\omega = x/x_p$ or $\bar r/R_p$ and $\phi = x_p$ or $R_p$ times $\sqrt{k/D_{TA}^{e}}$:
\end{itemize}
\begin{tabularx}{\linewidth}{@{}l X X X@{}}
\toprule
 & \textbf{Slab} & \textbf{Cylinder} & \textbf{Sphere}\\\\
\midrule
$\psi$ & $\dfrac{\cosh(\phi\omega)}{\cosh\phi}$ & $\dfrac{I_0(\phi\omega)}{I_0(\phi)}$ & $\dfrac{\sinh(\phi\omega)}{\omega\sinh\phi}$\\\\[10pt]
$\eta$ & $\dfrac{\tanh\phi}{\phi}$ & $\dfrac{2\,I_1(\phi)}{\phi\,I_0(\phi)}$ & $\dfrac{3}{\phi}\left[\dfrac{1}{\tanh\phi}-\dfrac{1}{\phi}\right]$\\\\
\bottomrule
\end{tabularx}
\notetext{$I_i$ is the modified Bessel function of order $i$.}
\begin{itemize}
\item \textbf{Non-isothermal pellet} (slide~46): with $\gamma = E/(RT_s)$ (Arrhenius number) and $\beta = \Delta T_{max}/T_s = -\Delta H_{Rx} D_e C_{As}/(k_s T_s)$, an exothermic reaction can run hotter inside the pellet than at its surface, and $\eta$ can \emph{exceed}~1.
\end{itemize}

\spsub{8.8 \enspace Falsified kinetics and the diagnostic criteria}
\begin{itemize}
\item \textbf{Falsified kinetics} (slide~45): cumene cracking on SiO$_2$--Al$_2$O$_3$, measured rate against $1/T$ for four pellet radii. Small pellets give the steep, true-activation-energy line; as the pellet grows the line drops and its slope falls toward $E/2$ --- the Arrhenius plot itself tells you when internal diffusion has taken over.
\item \textbf{Weisz--Prater} (slide~49): evaluate limitation from the \emph{observed} rate alone,
\end{itemize}
\begin{formulabox}
\begin{align*}
C_{WP} &= \eta\,\phi^{2} = \frac{(-r_A')_{\text{obs}}\,\rho_c\, R^{2}}{D_e\,C_{As}}
 = \frac{\text{actual reaction rate}}{\text{a diffusion rate}}\\[3pt]
C_{WP} &\ll 1 &&\text{no concentration gradients in the pellet}\\[3pt]
C_{WP} &\gg 1 &&\text{internal diffusion controls the reaction}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{Mears criteria} (slide~56) for the \emph{external} film:
\end{itemize}
\begin{formulabox}
\begin{align*}
\frac{(-r_A')\,\rho_b\, R\, n}{k_c\, C_{Ab}} &< 0.15
 &&\text{external concentration gradients negligible}\\[5pt]
\frac{\left|-\Delta H_{Rx}\,(-r_A')\,\rho_b\, R\, E\right|}{h\, T^{2}\, R_g} &< 0.15
 &&\text{external temperature gradients negligible}
\end{align*}
\end{formulabox}
\notetext{$n$ = order of reaction, $R$ = pellet radius, $\rho_b$ = bulk density, $h$ = heat-transfer coefficient between gas and pellet. Weisz--Prater is the practical internal test because it needs only the observed rate; Mears is its external counterpart.}

\spsub{8.9 \enspace External mass transfer and the overall effectiveness factor}
\begin{itemize}
\item \textbf{Mass-transfer coefficient} (slide~51): $k_c = 0.6\,\dfrac{D_{AB}^{2/3}}{\nu^{1/6}}\times\dfrac{U^{1/2}}{d_p^{1/2}}$. Plotting $-r_A'$ against $(U/d_p)^{1/2}$: the rate first rises with velocity (\textbf{diffusion limited}, the boundary layer thins) and then plateaus (\textbf{reaction limited}, the surface reaction is the bottleneck).
\item \textbf{Steady-state matching} (slides~53--54): the molar rate of transfer from the bulk to the external surface equals the net rate of reaction on and within the pellet, $M_{Ar} = -r_A''\,(\text{external area} + \text{internal area})$, with external area $= a_c\,\Delta V$ and internal area $= S_a\,\rho_b\,\Delta V$.
\end{itemize}
\begin{formulabox}
\begin{align*}
\Omega &= \frac{\text{actual overall rate}}{\text{rate if the whole surface saw } C_{Ab}, T_b}
 = \frac{\eta}{1 + \eta\, k_1''\, S_a\, \rho_b / (k_c\, a_c)}
\end{align*}
\end{formulabox}

\spsub{8.10 \enspace Packed beds: design with diffusion and reaction}
\begin{itemize}
\item The packed bed is the workhorse (slide~58) --- steam reforming, catalytic reforming, isomerization, polymerization, ammonia and SO$_3$ synthesis --- run adiabatic or non-adiabatic, particle sizes $1/8''$ to $1/4''$.
\item \textbf{Balance over a bed slice} (slide~60): $A_c W_{Az}\rvert_z - A_c W_{Az}\rvert_{z+\Delta z} + r_A'\,\rho_b\, A_c\,\Delta z = 0$, i.e.\ $-\dfrac{dW_{Az}}{dz} + r_A'\rho_b = 0$. With a dispersion term this becomes
\end{itemize}
\begin{formulabox}
\begin{align*}
D_a\,\frac{d^2 C_{Ab}}{dz^2} - U\,\frac{dC_{Ab}}{dz} - \Omega\,\rho_b\, k'' S_a\, C_{Ab} &= 0\\[5pt]
X = 1 - \frac{C_{Ab}}{C_{Ab0}} &= 1 - e^{-\left(\rho_b\, k''\, S_a\, \Omega\, L\right)/U}
 &&\text{dispersion neglected}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{Configurations} (slide~59): \textbf{multi-bed} (unit operations between PBRs); \textbf{reverse flow} --- for exothermic reactions the temperature rise along the bed can deactivate catalyst, so the flow direction is switched periodically, which also preheats the feed; \textbf{counter-flow} --- two chambers, a lean stream preheats the fluid and raises the rate; \textbf{radial flow} --- feed enters and flows radially through an annulus of catalyst, for operation under low pressure drop.
\end{itemize}

%================================================================== Non-ideal
\spsection{9 \enspace Non-Ideal Reactors and Residence-Time Distributions}
\notetext{Source: \texttt{Non\_ideal\_KRD.pdf} --- headed \emph{Reaction Engineering / CHE~G641}, the graduate treatment of this material.}

\spsub{9.1 \enspace Where ideality breaks}
\begin{itemize}
\item \textbf{Why it matters (slides 3--4):} the ideal design equations assume identical residence times (PFR/PBR, no axial diffusion) or perfect mixing (CSTR/batch); real vessels sit in between --- non-uniform flow, inefficient or absent mixing, bypassing and dead zones change conversion, yield and the temperature development of the process.
\item \textbf{CSTR (slide 5):} the feed entrance runs at relatively high concentration while stagnant \emph{dead zones} run low; part of the flow can short-circuit the vessel as bypass.
\item \textbf{PBR (slide 6):} ideal plug flow moves every molecule at the same axial rate; the real bed adds turbulent velocity profiles, back-mixing and molecular diffusion, so molecules travel at different speeds and in different directions (channeling, slide~14).
\item \textbf{Micro vs macro mixing (slides 8--9):} macromixing is the RTD itself; micromixing asks whether molecules of different ages meet at all --- \emph{complete segregation}, same-age globules leaving together, versus \emph{complete micromixing} --- and whether mixing occurs early or late along the flow path.
\end{itemize}

\spsub{9.2 \enspace The RTD functions}
\begin{formulabox}
\begin{align*}
E(t) &= \frac{C(t)}{\displaystyle\int_0^{\infty} C\,dt}
 &&\text{normalised tracer response}\\[6pt]
F(t) &= \int_0^{t} E\,dt
 &&\text{cumulative fraction younger than } t\\[5pt]
t_m &= \int_0^{\infty} t\,E\,dt = \frac{V}{v_0}
 &&\text{mean residence time}\\[5pt]
\sigma^{2} &= \int_0^{\infty} \left(t - t_m\right)^{2} E\,dt
 &&\text{variance, i.e.\ the spread}\\[5pt]
1-F(t) &= \frac{V}{v}\,I(t)
 &&\text{internal age distribution } I(t)
\end{align*}
\end{formulabox}
\notetext{Normalising by the area is what makes $E(t)$ a probability density, and it is the step most often missed when a pulse experiment is worked up by hand.}
\notetext{The profiles are a diagnostic (slides 14, 21): a sharp spike reads as a nearly ideal PFR, a decaying exponential as a nearly ideal CSTR, an early spike with a long tail as a PBR with channeling and dead zones, and a two-humped curve as a CSTR with dead zones.}

\spsub{9.3 \enspace Ideal limits}
\begin{formulabox}
\begin{align*}
\text{PFR:}\quad & E(t) = \delta\left(t-\tau\right)
 &&\text{all fluid elements the same age}\\[3pt]
\text{CSTR:}\quad & E(t) = \frac{e^{-t/\tau}}{\tau}
 &&\text{exponential; some fluid leaves almost at once}
\end{align*}
\end{formulabox}

\spsub{9.4 \enspace Zero-parameter models: segregation and maximum mixedness}
\begin{formulabox}
\begin{align*}
\bar{X} &= \int_0^{\infty} X_{\text{batch}}(t)\,E(t)\,dt
 &&\text{each fluid element is a small batch reactor}
\end{align*}
\end{formulabox}
\begin{itemize}
\item \textbf{$n>1$:} $\bar{X}$ \emph{exceeds} the ideal-mixed conversion. \textbf{$0<n<1$:} the opposite. Complete segregation is therefore a bound, and it tells you which way a real reactor departs from ideal mixing without any further modelling.
\end{itemize}
\notetext{The deck's model menu (slide~22): \emph{zero} adjustable parameters --- the segregation model above and the maximum-mixedness model, its micromixed opposite extreme; \emph{one} parameter --- tanks-in-series and dispersion; \emph{two} parameters --- the real vessel as a combination of ideal reactors (\S9.7).}

\spsub{9.5 \enspace Tanks-in-series model}
\begin{formulabox}
\begin{align*}
E(t) &= \frac{t^{\,n-1}}{(n-1)!\;\tau_i^{\,n}}\;e^{-t/\tau_i}
 &&\text{gamma distribution}\\[5pt]
n &= \frac{t_m^{2}}{\sigma^{2}}
 &&\text{number of tanks, from the tracer data alone}\\[5pt]
X &= 1 - \left(1 + k\tau_i\right)^{-n}
 &&\text{first-order conversion}
\end{align*}
\end{formulabox}
\notetext{Where the gamma form comes from (slides 26--28): a pulse through $N$ equal CSTRs gives $C_1 = C_0e^{-t/\tau_i}$, $C_2 = C_0\,t\,e^{-t/\tau_i}/\tau_i$, $C_3 = C_0\,t^2e^{-t/\tau_i}/2\tau_i^2$, and in general $E(t) = t^{N-1}e^{-t/\tau_i}/[(N-1)!\,\tau_i^{N}]$, or dimensionless $E(\theta) = N(N\theta)^{N-1}e^{-N\theta}/(N-1)!$. As $N\to\infty$ the curve collapses to the plug-flow spike, which is why $n = t_m^2/\sigma^2$ measures the departure from plug flow.}

\spsub{9.6 \enspace Dispersion model}
\begin{formulabox}
\begin{align*}
X &= 1 - \frac{4q\,e^{Pe/2}}
 {(1+q)^{2}e^{Pe\,q/2} - (1-q)^{2}e^{-Pe\,q/2}}
 &&\text{closed vessel, first order}\\[6pt]
q &= \sqrt{1 + \frac{4\,Da}{Pe}},
 &\qquad Da &= k\tau,
 &\qquad Pe &= \frac{UL}{D_a}\\[5pt]
\sigma_{\theta}^{2} &= \frac{2}{Pe} - \frac{2}{Pe^{2}}\left(1-e^{-Pe}\right)
 &&\text{link from tracer data to } Pe\\[5pt]
n &= \frac{Pe}{2} + 1
 &&\text{equivalent number of tanks}
\end{align*}
\end{formulabox}
\notetext{The picture behind it (slides 29--33): molecules diffuse ahead of and lag behind the molar average velocity, i.e.\ a Fickian dispersion $D_a\,dC/dz$ superimposed on the bulk flow, so the tracer pulse broadens as it travels; the mole balance over a slice $\Delta Z$ with $F_A = uAC_A + AD_a\,dC_A/dz$ yields the closed-vessel solution above. The single parameter $D_a/(UL)$ is read from a pulse tracer test; the deck's $E$-curves for $D_a/(UL) = 0.002,\ 0.025,\ 0.2$ show small, intermediate and large deviation from plug flow.}

\spsub{9.7 \enspace Two-parameter models}
\begin{itemize}
\item A real CSTR is represented as an \textbf{ideal reacting region} of volume $V_s = \alpha V$ with a \textbf{dead zone} $V_d = (1-\alpha)V$, plus a \textbf{bypass} stream; $\beta = v_s/v_0$ is the fraction of the feed entering the reacting region (slides 40--41). The modelled conversion is
\[
1-X = \beta + \frac{(1-\beta)^2}{1-\beta+\alpha\tau k}.
\]
\item Tracer response takes the form
\[
\frac{C_T}{C_{T0}} = 1 - \left(1-\beta\right)e^{-(1-\beta)t/(\alpha\tau)}
\]
and linearising it (slide~42),
\[
\ln\frac{C_{T0}}{C_{T0}-C_T} = \ln\frac{1}{1-\beta} + \frac{1-\beta}{\alpha}\cdot\frac{t}{\tau},
\]
so the intercept and slope of the tracer straight line give $(\alpha,\beta)$.
\item The deck's second two-parameter variant (slides 43--44) models a real CSTR as \textbf{two ideal CSTRs with interchange} --- a highly agitated region near the impeller exchanging with a quieter one, $V_d = (1-\alpha)V$ --- with
\[
X = \frac{(\beta+\alpha\tau k)(\beta+(1-\alpha)\tau k)-\beta^{2}}{(1+\beta+\alpha\tau k)(\beta+(1-\alpha)\tau k)-\beta^{2}},
\]
and $(\alpha,\beta)$ again fixed by a tracer test, the response now a sum of two exponentials with exponents $m_{1},m_{2}$.
\end{itemize}

\spsub{9.8 \enspace Worked from the deck}
\begin{itemize}
\item \textbf{Pulse tracer} (slide~17):
\[

\qquad\Longrightarrow\qquad \boxed{\;\text{fraction between 3 and 6 min} = 0.513\;}
\]
\notetext{Just over half the fluid leaves within a minute either side of the mean --- the spread is the whole story here, not the mean.}
\item \textbf{Dispersion example} (slides~52--59):
\[
Pe = 7.74,\qquad Da = 1.28
\]
\end{itemize}
